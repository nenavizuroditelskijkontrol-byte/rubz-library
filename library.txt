--==============================================================================
-- ХЕЛПЕР: Create — создать Instance с параметрами
--==============================================================================
-- Универсальная обёртка над Instance.new
-- props — таблица свойств, включая Parent (специально обрабатывается последним)
local function Create(class, props)
    local inst = Instance.new(class)
    for k, v in pairs(props or {}) do
        if k ~= "Parent" then
            inst[k] = v
        end
    end
    if props and props.Parent then
        inst.Parent = props.Parent
    end
    return inst
end

--==============================================================================
-- ХЕЛПЕР: Corner — быстро накинуть UICorner
--==============================================================================
local function Corner(inst, r)
    return Create("UICorner", {
        CornerRadius = UDim.new(0, r or 6),
        Parent = inst,
    })
end

--==============================================================================
-- ХЕЛПЕР: Stroke — быстро накинуть UIStroke
--==============================================================================
local function Stroke(inst, color, thick, trans)
    return Create("UIStroke", {
        Color = color or T.Border,
        Thickness = thick or 1,
        Transparency = trans or 0,
        ApplyStrokeMode = Enum.ApplyStrokeMode.Border,
        Parent = inst,
    })
end

--==============================================================================
-- ХЕЛПЕР: Gradient — быстро накинуть UIGradient
--==============================================================================
local function Gradient(inst, color1, color2, rotation)
    return Create("UIGradient", {
        Color = ColorSequence.new(color1, color2 or color1),
        Rotation = rotation or 0,
        Parent = inst,
    })
end

--==============================================================================
-- ХЕЛПЕР: Tween — создать и запустить анимацию
--==============================================================================
local function Tween(inst, time, props, style, dir)
    local anim = TweenService:Create(
        inst,
        TweenInfo.new(
            time or 0.15,
            style or Enum.EasingStyle.Quad,
            dir or Enum.EasingDirection.Out
        ),
        props
    )
    anim:Play()
    return anim
end

--==============================================================================
-- ХЕЛПЕР: Round — округление в зависимости от размера диапазона
--==============================================================================
local function Round(v, range)
    if range <= 5 then
        return math.floor(v * 100 + 0.5) / 100
    end
    return math.floor(v + 0.5)
end

--==============================================================================
-- ХЕЛПЕР: GetGuiParent — получить контейнер для ScreenGui
--==============================================================================
-- Пытаемся сначала gethui (в executor'ах), потом CoreGui, потом PlayerGui
local function GetGuiParent()
    if gethui then
        local ok, hui = pcall(gethui)
        if ok and hui then return hui end
    end
    local ok, cg = pcall(function() return CoreGui end)
    if ok and cg then
        local canUse = pcall(function() return cg.Name end)
        if canUse then return cg end
    end
    return LocalPlayer:WaitForChild("PlayerGui")
end

--==============================================================================
-- ХЕЛПЕР: MakeDraggable — сделать окно перетаскиваемым
--==============================================================================
local function MakeDraggable(frame, handle)
    handle = handle or frame
    local dragging, dragStart, startPos

    handle.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            dragStart = input.Position
            startPos = frame.Position
        end
    end)

    UserInputService.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch then
            dragging = false
        end
    end)

    UserInputService.InputChanged:Connect(function(input)
        if not dragging then return end
        if input.UserInputType == Enum.UserInputType.MouseMovement
        or input.UserInputType == Enum.UserInputType.Touch then
            local d = input.Position - dragStart
            frame.Position = UDim2.new(
                startPos.X.Scale, startPos.X.Offset + d.X,
                startPos.Y.Scale, startPos.Y.Offset + d.Y
            )
        end
    end)
end

--==============================================================================
-- ХЕЛПЕР: GetKeyName — имя нажатой клавиши/кнопки мыши
--==============================================================================
local function GetKeyName(input)
    if input.UserInputType == Enum.UserInputType.Keyboard then
        return input.KeyCode.Name
    elseif input.UserInputType == Enum.UserInputType.MouseButton1 then
        return "Mouse1"
    elseif input.UserInputType == Enum.UserInputType.MouseButton2 then
        return "Mouse2"
    elseif input.UserInputType == Enum.UserInputType.MouseButton3 then
        return "Mouse3"
    end
    return nil
end

--==============================================================================
-- ХЕЛПЕР: AddShadow — быстрая мягкая тень под элементом
--==============================================================================
local function AddShadow(parent, size, position, transparency)
    local shadow = Create("Frame", {
        Size = size or UDim2.new(1, -4, 1, -4),
        Position = position or UDim2.new(0, 2, 0, 2),
        BackgroundColor3 = Color3.new(0, 0, 0),
        BackgroundTransparency = transparency or 0.75,
        BorderSizePixel = 0,
        ZIndex = (parent.ZIndex or 1) - 1,
        Parent = parent,
    })
    Corner(shadow, 10)
    return shadow
end

--/////////////////////////////////////////////////////////////////////////////
-- EFFECT: Ripple (волна от клика на кнопке)
--/////////////////////////////////////////////////////////////////////////////
local function AddRipple(button)
    button.ClipsDescendants = true
    button.MouseButton1Down:Connect(function()
        local m = UserInputService:GetMouseLocation()
        local relX = m.X - button.AbsolutePosition.X
        local relY = m.Y - button.AbsolutePosition.Y
        local r = Create("Frame", {
            Size = UDim2.fromOffset(8, 8),
            Position = UDim2.fromOffset(relX - 4, relY - 4),
            BackgroundColor3 = Color3.new(1, 1, 1),
            BackgroundTransparency = 0.65,
            BorderSizePixel = 0,
            ZIndex = (button.ZIndex or 1) + 10,
            Parent = button,
        })
        Corner(r, 999)
        local size = math.max(button.AbsoluteSize.X, button.AbsoluteSize.Y) * 2.5
        Tween(r, 0.7, {
            Size = UDim2.fromOffset(size, size),
            Position = UDim2.fromOffset(relX - size / 2, relY - size / 2),
            BackgroundTransparency = 1,
        }, Enum.EasingStyle.Quart)
        task.delay(0.75, function()
            if r and r.Parent then
                r:Destroy()
            end
        end)
    end)
end

--/////////////////////////////////////////////////////////////////////////////
-- EFFECT: MultiGlow (три обводки = glow)
--/////////////////////////////////////////////////////////////////////////////
local function AddMultiGlow(frame, color, intensity)
    intensity = intensity or 1
    local strokes = {}
    table.insert(strokes, Stroke(frame, color, 1, 0.25 / intensity))
    table.insert(strokes, Stroke(frame, color, 2, 0.7 / intensity))
    table.insert(strokes, Stroke(frame, color, 3, 0.85 / intensity))
    return strokes
end

local function CreateStarField(parent, count)
    count = count or 30
    local holder = Create("Frame", {
        Size = UDim2.fromScale(1, 1),
        BackgroundTransparency = 1,
        ZIndex = 1,
        ClipsDescendants = true,
        Parent = parent,
    })
    local stars = {}
    for i = 1, count do
        local sz = math.random(2, 4)
        local star = Create("Frame", {
            Size = UDim2.fromOffset(sz, sz),
            Position = UDim2.new(math.random(), 0, math.random(), 0),
            BackgroundColor3 = Color3.new(1, 1, 1),
            BackgroundTransparency = 0.3 + math.random() * 0.5,
            BorderSizePixel = 0,
            ZIndex = 1,
            Parent = holder,
        })
        Corner(star, 999)
        stars[i] = {
            Frame = star,
            baseX = star.Position.X.Scale,
            baseY = star.Position.Y.Scale,
            speedY = 0.005 + math.random() * 0.02,
            speedX = (math.random() - 0.5) * 0.01,
            phase = math.random() * math.pi * 2,
            twinkle = 0.3 + math.random() * 0.5,
        }
    end
    local conn = RunService.RenderStepped:Connect(function(_, dt)
        if not holder or not holder.Parent then return end
        for _, s in ipairs(stars) do
            if not s.Frame or not s.Frame.Parent then continue end
            s.baseY = s.baseY + s.speedY * dt * 10
            if s.baseY > 1 then s.baseY = -0.05 end
            s.baseX = s.baseX + s.speedX * dt
            if s.baseX < 0 then s.baseX = 1 elseif s.baseX > 1 then s.baseX = 0 end
            s.Frame.Position = UDim2.new(s.baseX, 0, s.baseY, 0)
            local pulse = (math.sin(tick() * 2 + s.phase) + 1) / 2
            s.Frame.BackgroundTransparency = 0.3 + pulse * s.twinkle
        end
    end)
    return holder, conn
end

--/////////////////////////////////////////////////////////////////////////////
-- EFFECT: HoverGlow (плавное усиление свечения при наведении)
--/////////////////////////////////////////////////////////////////////////////
local function AddHoverGlow(button, glowColor, baseTrans, hoverTrans)
    glowColor = glowColor or T.PrimaryLight
    baseTrans = baseTrans or 0.8
    hoverTrans = hoverTrans or 0.3

    local glow = Create("Frame", {
        Size = UDim2.new(1, 6, 1, 6),
        Position = UDim2.new(0, -3, 0, -3),
        BackgroundColor3 = glowColor,
        BackgroundTransparency = 1,
        BorderSizePixel = 0,
        ZIndex = (button.ZIndex or 1) - 1,
        Parent = button.Parent,
    })
    Corner(glow, 14)

    button.MouseEnter:Connect(function()
        Tween(glow, 0.2, { BackgroundTransparency = hoverTrans })
    end)
    button.MouseLeave:Connect(function()
        Tween(glow, 0.2, { BackgroundTransparency = 1 })
    end)
    return glow
end

--/////////////////////////////////////////////////////////////////////////////
-- EFFECT: Pulse (пульсация любого элемента)
--/////////////////////////////////////////////////////////////////////////////
local function AddPulse(inst, prop, from, to, duration)
    local pulseConn
    pulseConn = RunService.RenderStepped:Connect(function()
        if not inst or not inst.Parent then
            if pulseConn then pulseConn:Disconnect() end
            return
        end
        local t = (math.sin(tick() * (1 / (duration or 2)) * math.pi * 2) + 1) / 2
        local val = from + (to - from) * t
        inst[prop] = val
    end)
    return pulseConn
end

--/////////////////////////////////////////////////////////////////////////////
-- Library:Debounce(key, delay) — дебаунсер (не даёт вызывать чаще delay)
--/////////////////////////////////////////////////////////////////////////////
local _debounce = {}
function Library:Debounce(key, delay)
    delay = delay or 1
    local now = tick()
    if _debounce[key] and (now - _debounce[key]) < delay then
        return false
    end
    _debounce[key] = now
    return true
end

--/////////////////////////////////////////////////////////////////////////////
-- Library:Throttle(key, delay) — то же, что Debounce, но наоборот
--/////////////////////////////////////////////////////////////////////////////
local _throttle = {}
function Library:Throttle(key, delay)
    delay = delay or 0.1
    local now = tick()
    if _throttle[key] and (now - _throttle[key]) < delay then
        return true
    end
    _throttle[key] = now
    return false
end

--/////////////////////////////////////////////////////////////////////////////
-- Library:FormatNumber(n) — форматировать число (1000 → "1K")
--/////////////////////////////////////////////////////////////////////////////
function Library:FormatNumber(n)
    if n >= 1e9 then return string.format("%.1fB", n / 1e9) end
    if n >= 1e6 then return string.format("%.1fM", n / 1e6) end
    if n >= 1e3 then return string.format("%.1fK", n / 1e3) end
    return tostring(n)
end

--/////////////////////////////////////////////////////////////////////////////
-- Library:FormatTime(seconds) — секунды в "1:23:45"
--/////////////////////////////////////////////////////////////////////////////
function Library:FormatTime(seconds)
    local h = math.floor(seconds / 3600)
    local m = math.floor((seconds % 3600) / 60)
    local s = math.floor(seconds % 60)
    if h > 0 then
        return string.format("%d:%02d:%02d", h, m, s)
    end
    return string.format("%d:%02d", m, s)
end

--/////////////////////////////////////////////////////////////////////////////
-- Library:DeepCopy(tbl) — глубокая копия таблицы
--/////////////////////////////////////////////////////////////////////////////
function Library:DeepCopy(tbl)
    local copy = {}
    for k, v in pairs(tbl) do
        if type(v) == "table" then
            copy[k] = Library:DeepCopy(v)
        else
            copy[k] = v
        end
    end
    return copy
end

--/////////////////////////////////////////////////////////////////////////////
-- Library:Merge(t1, t2) — слить две таблицы (t2 перезаписывает t1)
--/////////////////////////////////////////////////////////////////////////////
function Library:Merge(t1, t2)
    local result = Library:DeepCopy(t1 or {})
    for k, v in pairs(t2 or {}) do
        result[k] = v
    end
    return result
end

local function CreateTab(Window, tabName)
    local env      = Window._env
    local Create   = env.Create
    local Corner   = env.Corner
    local Stroke   = env.Stroke
    local Tween    = env.Tween
    local Icons    = env.Icons
    local SelectTab = env.SelectTab

    local Tab = {}
    Tab.Name = tabName
    Tab.Sections = {}
    Tab.Elements = {}

    --//////////////////////////////////////////////////////////////////////////
    -- КНОПКА ВКЛАДКИ В САЙДБАРЕ
    --//////////////////////////////////////////////////////////////////////////
    local Button = Create("TextButton", {
        Size = UDim2.new(1, 0, 0, 40),
        BackgroundColor3 = T.Bg3,
        BackgroundTransparency = 1,
        Text = "",
        AutoButtonColor = false,
        BorderSizePixel = 0,
        ClipsDescendants = true,
        ZIndex = 16,
        Parent = Window.TabList,
    })
    Corner(Button, 10)

    local ButtonStroke = Stroke(Button, T.BorderSoft, 1, 1)

    -- иконка-контейнер
    local IconHolder = Create("Frame", {
        Size = UDim2.fromOffset(16, 16),
        Position = UDim2.new(0, 14, 0.5, -8),
        BackgroundTransparency = 1,
        ZIndex = 17,
        Parent = Button,
    })

    local IconParts = {}
    table.insert(IconParts, Icons.Dot(IconHolder, 8, T.TextDim, 18))

    -- индикатор слева
    local Indicator = Create("Frame", {
        Size = UDim2.new(0, 3, 0, 12),
        Position = UDim2.new(0, 4, 0.5, -6),
        BackgroundColor3 = T.PrimaryLight,
        BackgroundTransparency = 1,
        BorderSizePixel = 0,
        ZIndex = 18,
        Parent = Button,
    })
    Corner(Indicator, 2)

    -- текст вкладки
    local TextLabel = Create("TextLabel", {
        Size = UDim2.new(1, -48, 1, 0),
        Position = UDim2.new(0, 36, 0, 0),
        BackgroundTransparency = 1,
        Text = tabName,
        TextColor3 = T.TextDim,
        TextSize = 14,
        Font = Enum.Font.GothamMedium,
        TextXAlignment = Enum.TextXAlignment.Left,
        ZIndex = 17,
        Parent = Button,
    })

    --//////////////////////////////////////////////////////////////////////////
    -- СТРАНИЦА ВКЛАДКИ
    --//////////////////////////////////////////////////////////////////////////
    local Page = Create("CanvasGroup", {
        Size = UDim2.new(1, 0, 1, 0),
        BackgroundTransparency = 1,
        GroupTransparency = 0,
        Visible = false,
        Parent = Window.Content,
    })

    local Scroll = Create("ScrollingFrame", {
        Size = UDim2.new(1, 0, 1, 0),
        BackgroundTransparency = 1,
        BorderSizePixel = 0,
        ScrollBarThickness = 4,
        ScrollBarImageColor3 = T.Primary,
        ScrollBarImageTransparency = 0.5,
        CanvasSize = UDim2.new(0, 0, 0, 0),
        AutomaticCanvasSize = Enum.AutomaticSize.Y,
        Parent = Page,
    })
    Create("UIPadding", {
        PaddingTop = UDim.new(0, 2),
        PaddingBottom = UDim.new(0, 14),
        PaddingRight = UDim.new(0, 10),
        Parent = Scroll,
    })
    Create("UIListLayout", {
        Padding = UDim.new(0, 12),
        SortOrder = Enum.SortOrder.LayoutOrder,
        Parent = Scroll,
    })

    Tab.Button = Button
    Tab.ButtonStroke = ButtonStroke
    Tab.TextLabel = TextLabel
    Tab.Indicator = Indicator
    Tab.IconHolder = IconHolder
    Tab.IconParts = IconParts
    Tab.Page = Page
    Tab.Scroll = Scroll

    --//////////////////////////////////////////////////////////////////////////
    -- МЕТОД: Установить иконку вкладки
    --//////////////////////////////////////////////////////////////////////////
    function Tab:SetIcon(drawFunc)
        for _, p in ipairs(Tab.IconParts) do
            if p and p.Parent then
                p:Destroy()
            end
        end
        Tab.IconParts = {}
        local r = drawFunc and drawFunc(IconHolder, T.TextDim, 18)
        if type(r) == "table" then
            for _, p in ipairs(r) do
                table.insert(Tab.IconParts, p)
            end
        elseif r then
            table.insert(Tab.IconParts, r)
        end
    end

    --//////////////////////////////////////////////////////////////////////////
    -- ХОВЕРЫ ВКЛАДКИ
    --//////////////////////////////////////////////////////////////////////////
    Button.MouseButton1Click:Connect(function()
        SelectTab(Tab)
    end)

    Button.MouseEnter:Connect(function()
        local cur = Window._getCurrentTab()
        if cur ~= Tab then
            Tween(Button, 0.18, { BackgroundTransparency = 0.6, BackgroundColor3 = T.Bg3 })
            Tween(TextLabel, 0.18, { TextColor3 = T.Text })
            for _, ic in ipairs(Tab.IconParts) do
                Tween(ic, 0.18, { BackgroundColor3 = T.PrimaryLight })
            end
        end
    end)
    Button.MouseLeave:Connect(function()
        local cur = Window._getCurrentTab()
        if cur ~= Tab then
            Tween(Button, 0.18, { BackgroundTransparency = 1 })
            Tween(TextLabel, 0.18, { TextColor3 = T.TextDim })
            for _, ic in ipairs(Tab.IconParts) do
                Tween(ic, 0.18, { BackgroundColor3 = T.TextDim })
            end
        end
    end)

        --//////////////////////////////////////////////////////////////////////
        -- ЭЛЕМЕНТ: Toggle (переключатель с пружинкой)
        --//////////////////////////////////////////////////////////////////////
        function Section:Toggle(text, default, callback, flag)
            local state = default and true or false
            local Container = Create("Frame", {
                Size = UDim2.new(1, 0, 0, 36),
                BackgroundTransparency = 1,
                Parent = Frame,
            })
            Create("TextLabel", {
                Size = UDim2.new(1, -70, 1, 0),
                BackgroundTransparency = 1,
                Text = text,
                TextColor3 = T.Text,
                TextSize = 14,
                Font = Enum.Font.Gotham,
                TextXAlignment = Enum.TextXAlignment.Left,
                Parent = Container,
            })

            local Switch = Create("Frame", {
                Size = UDim2.fromOffset(46, 24),
                Position = UDim2.new(1, -46, 0.5, -12),
                BackgroundColor3 = state and T.Primary or T.Bg3,
                BackgroundTransparency = state and 0 or Library.Opacity.Element,
                BorderSizePixel = 0,
                Parent = Container,
            })
            Corner(Switch, 12)
            local SwitchStroke = Stroke(Switch,
                state and T.PrimaryLight or T.Border, 1,
                state and 0.2 or 0.6)

            local Knob = Create("Frame", {
                Size = UDim2.fromOffset(18, 18),
                Position = state and UDim2.new(1, -21, 0.5, -9) or UDim2.new(0, 3, 0.5, -9),
                BackgroundColor3 = Color3.new(1, 1, 1),
                BorderSizePixel = 0,
                Parent = Switch,
            })
            Corner(Knob, 999)

            local Hit = Create("TextButton", {
                Size = UDim2.new(1, 0, 1, 0),
                BackgroundTransparency = 1,
                Text = "",
                Parent = Container,
            })

            local function set(value, silent)
                state = value and true or false
                Tween(Switch, 0.25, {
                    BackgroundColor3 = state and T.Primary or T.Bg3,
                    BackgroundTransparency = state and 0 or Library.Opacity.Element,
                }, Enum.EasingStyle.Quart)
                Tween(Knob, 0.32, {
                    Position = state and UDim2.new(1, -21, 0.5, -9) or UDim2.new(0, 3, 0.5, -9),
                }, Enum.EasingStyle.Back)
                Tween(SwitchStroke, 0.22, {
                    Color = state and T.PrimaryLight or T.Border,
                    Transparency = state and 0.2 or 0.6,
                })
                if flag then Library.Flags[flag] = state end
                if not silent and callback then task.spawn(callback, state) end
            end

            Hit.MouseButton1Click:Connect(function() set(not state) end)
            if flag then Library.Flags[flag] = state end

            return {
                Set = function(_, v) set(v, true) end,
                Get = function() return state end,
                Instance = Container,
            }
        end

        --//////////////////////////////////////////////////////////////////////
        -- ЭЛЕМЕНТ: Slider (ползунок с glow и градиентом)
        --//////////////////////////////////////////////////////////////////////
        function Section:Slider(text, min, max, default, callback, flag)
            local value = default or min
            local range = max - min

            local Container = Create("Frame", {
                Size = UDim2.new(1, 0, 0, 50),
                BackgroundTransparency = 1,
                Parent = Frame,
            })
            Create("TextLabel", {
                Size = UDim2.new(1, -90, 0, 18),
                BackgroundTransparency = 1,
                Text = text,
                TextColor3 = T.Text,
                TextSize = 14,
                Font = Enum.Font.Gotham,
                TextXAlignment = Enum.TextXAlignment.Left,
                Parent = Container,
            })
            local ValueLabel = Create("TextLabel", {
                Size = UDim2.fromOffset(90, 18),
                Position = UDim2.new(1, -90, 0, 0),
                BackgroundTransparency = 1,
                Text = tostring(value),
                TextColor3 = T.PrimaryLight,
                TextSize = 13,
                Font = Enum.Font.GothamBold,
                TextXAlignment = Enum.TextXAlignment.Right,
                Parent = Container,
            })
            local Bar = Create("Frame", {
                Size = UDim2.new(1, 0, 0, 6),
                Position = UDim2.new(0, 0, 0, 36),
                BackgroundColor3 = T.Bg3,
                BackgroundTransparency = Library.Opacity.Element,
                BorderSizePixel = 0,
                Parent = Container,
            })
            Corner(Bar, 3)
            Stroke(Bar, T.Border, 1, 0.65)

            local Fill = Create("Frame", {
                Size = UDim2.new(0, 0, 1, 0),
                BackgroundColor3 = Color3.new(1, 1, 1),
                BorderSizePixel = 0,
                Parent = Bar,
            })
            Corner(Fill, 3)
            Create("UIGradient", {
                Color = ColorSequence.new({
                    ColorSequenceKeypoint.new(0, T.PrimaryDark),
                    ColorSequenceKeypoint.new(0.5, T.Primary),
                    ColorSequenceKeypoint.new(1, T.PrimaryLight),
                }),
                Parent = Fill,
            })

            local Knob = Create("Frame", {
                Size = UDim2.fromOffset(16, 16),
                AnchorPoint = Vector2.new(0.5, 0.5),
                Position = UDim2.new(0, 0, 0.5, 0),
                BackgroundColor3 = Color3.new(1, 1, 1),
                BorderSizePixel = 0,
                ZIndex = 2,
                Parent = Bar,
            })
            Corner(Knob, 999)
            Stroke(Knob, T.PrimaryLight, 2, 0.15)

            local KnobGlow = Create("Frame", {
                Size = UDim2.fromOffset(26, 26),
                AnchorPoint = Vector2.new(0.5, 0.5),
                Position = UDim2.new(0, 0, 0.5, 0),
                BackgroundColor3 = T.PrimaryLight,
                BackgroundTransparency = 0.75,
                BorderSizePixel = 0,
                ZIndex = 1,
                Parent = Bar,
            })
            Corner(KnobGlow, 999)

            local Hit = Create("TextButton", {
                Size = UDim2.new(1, 0, 1, 0),
                BackgroundTransparency = 1,
                Text = "",
                Parent = Container,
            })

            local function apply(alpha, silent)
                alpha = math.clamp(alpha, 0, 1)
                value = env.Round(min + range * alpha, range)
                Fill.Size = UDim2.new(alpha, 0, 1, 0)
                Knob.Position = UDim2.new(alpha, 0, 0.5, 0)
                KnobGlow.Position = UDim2.new(alpha, 0, 0.5, 0)
                ValueLabel.Text = tostring(value)
                if flag then Library.Flags[flag] = value end
                if not silent and callback then task.spawn(callback, value) end
            end
            local function fromInput(input)
                apply((input.Position.X - Bar.AbsolutePosition.X) / Bar.AbsoluteSize.X)
            end

            local dragging = false
            Hit.InputBegan:Connect(function(input)
                if input.UserInputType == Enum.UserInputType.MouseButton1
                or input.UserInputType == Enum.UserInputType.Touch then
                    dragging = true
                    fromInput(input)
                    Tween(Knob, 0.14, { Size = UDim2.fromOffset(20, 20) }, Enum.EasingStyle.Back)
                    Tween(KnobGlow, 0.14, { Size = UDim2.fromOffset(32, 32), BackgroundTransparency = 0.6 })
                    Tween(Bar, 0.14, { Size = UDim2.new(1, 0, 0, 8), Position = UDim2.new(0, 0, 0, 35) })
                end
            end)
            table.insert(Window._conns, UserInputService.InputChanged:Connect(function(input)
                if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement
                or input.UserInputType == Enum.UserInputType.Touch) then
                    fromInput(input)
                end
            end))
            table.insert(Window._conns, UserInputService.InputEnded:Connect(function(input)
                if input.UserInputType == Enum.UserInputType.MouseButton1
                or input.UserInputType == Enum.UserInputType.Touch then
                    if dragging then
                        dragging = false
                        Tween(Knob, 0.2, { Size = UDim2.fromOffset(16, 16) }, Enum.EasingStyle.Back)
                        Tween(KnobGlow, 0.2, { Size = UDim2.fromOffset(26, 26), BackgroundTransparency = 0.75 })
                        Tween(Bar, 0.2, { Size = UDim2.new(1, 0, 0, 6), Position = UDim2.new(0, 0, 0, 36) })
                    end
                end
            end))

            do
                local alpha = range > 0 and (value - min) / range or 0
                Fill.Size = UDim2.new(alpha, 0, 1, 0)
                Knob.Position = UDim2.new(alpha, 0, 0.5, 0)
                KnobGlow.Position = UDim2.new(alpha, 0, 0.5, 0)
                ValueLabel.Text = tostring(value)
                if flag then Library.Flags[flag] = value end
            end

            return {
                Set = function(_, v)
                    local alpha = range > 0 and (v - min) / range or 0
                    apply(alpha, true)
                end,
                Get = function() return value end,
                Instance = Container,
            }
        end

                for _, b in ipairs(optionButtons) do
                    if b and b.Parent then b:Destroy() end
                end
                optionButtons = {}
            end

            local function renderList(list)
                clearOptions()
                for _, opt in ipairs(list) do
                    local OB = Create("TextButton", {
                        Size = UDim2.new(1, -10, 0, 30),
                        Position = UDim2.new(0, 5, 0, 0),

                        if callback then task.spawn(callback, opt) end

                    end)
                    table.insert(optionButtons, OB)
                end
            end

            local function close()
                if not open then return end
                open = false
                Tween(Container, 0.25, { Size = UDim2.new(1, 0, 0, 38) })
                Tween(Chev, 0.25, { Rotation = 0 })
                Tween(CS, 0.25, { Color = T.Border })
                SearchBox.Visible = false
                SearchBox.Text = ""
                renderList(allOptions)
            end

            local function openList()
                if open then return end
                open = true
                local hasSearch = #allOptions > 6
                SearchBox.Visible = hasSearch
                local h = math.min(#allOptions, 6) * 33 + 6 + (hasSearch and 32 or 0)
                Tween(Container, 0.28, { Size = UDim2.new(1, 0, 0, 38 + h) }, Enum.EasingStyle.Quart)

                Tween(CS, 0.25, { Color = T.PrimaryLight })
            end

                local q = string.lower(SearchBox.Text)
                if q == "" then
                    renderList(allOptions)
                else
                    local filtered = {}
                    for _, o in ipairs(allOptions) do
                        if string.find(string.lower(tostring(o)), q, 1, true) then
                            table.insert(filtered, o)
                        end
                    end
                    renderList(filtered)
                end
            end)

            renderList(options)

            Header.MouseButton1Click:Connect(function()
                if open then close() else openList() end
            end)
            if flag then Library.Flags[flag] = selected end

            return {
                Set = function(_, v)
                    selected = v
                    SelLabel.Text = tostring(v)
                    if flag then Library.Flags[flag] = v end
                end,
                Get = function() return selected end,
                Refresh = function(_, newOpts)
                    allOptions = newOpts or allOptions
                    renderList(allOptions)
                end,
                Instance = Container,
            }
        end

            local currentKey = default or "None"

                Size = UDim2.new(1, 0, 0, 36),

                Parent = Frame,

            local KeyBtn = Create("TextButton", {
                Size = UDim2.fromOffset(90, 30),
                Position = UDim2.new(1, -90, 0.5, -15),
                BackgroundColor3 = T.Bg3,
                BackgroundTransparency = Library.Opacity.Element,
                Text = tostring(currentKey),
                TextColor3 = T.Text,
                TextSize = 13,
                Font = Enum.Font.GothamMedium,
                AutoButtonColor = false,
                BorderSizePixel = 0,
                Parent = Container,
            })
            Corner(KeyBtn, 10)
            local KS = Stroke(KeyBtn, T.Border, 1, 0.6)
            if flag then Library.Flags[flag] = currentKey end

            KeyBtn.MouseButton1Click:Connect(function()
                if listening then return end
                listening = true
                KeyBtn.Text = "..."
                Tween(KeyBtn, 0.15, { BackgroundColor3 = T.Primary, BackgroundTransparency = 0.2 })
                Tween(KS, 0.15, { Color = T.PrimaryLight, Transparency = 0 })
            end)

            table.insert(Window._conns, UserInputService.InputBegan:Connect(function(input, gpe)
                if listening then
                    if input.UserInputType == Enum.UserInputType.Keyboard
                    or input.UserInputType == Enum.UserInputType.MouseButton1
                    or input.UserInputType == Enum.UserInputType.MouseButton2
                    or input.UserInputType == Enum.UserInputType.MouseButton3 then
                        local n = env.GetKeyName(input) or "None"
                        if n == "Escape" then n = "None" end
                        currentKey = n
                        KeyBtn.Text = n
                        listening = false
                        Tween(KeyBtn, 0.15, { BackgroundColor3 = T.Bg3, BackgroundTransparency = Library.Opacity.Element })
                        Tween(KS, 0.15, { Color = T.Border, Transparency = 0.6 })
                        if flag then Library.Flags[flag] = currentKey end
                        if callback then task.spawn(callback, currentKey) end
                    end
                    return
                end
                if gpe then return end
                if currentKey == "None" then return end
                local n = env.GetKeyName(input)
                if n and n == currentKey then
                    if callback then task.spawn(callback, currentKey) end
                end
            end))

            return {
                Set = function(_, v)
                    currentKey = v
                    KeyBtn.Text = tostring(v)
                    if flag then Library.Flags[flag] = v end
                end,
                Get = function() return currentKey end,
                Instance = Container,
            }
        end

                if not fromHex then
                    currentColor = Color3.fromHSV(hue, sat, val)

                end
                Preview.BackgroundColor3 = currentColor
                SV.BackgroundColor3 = Color3.fromHSV(hue, 1, 1)
                SVKnob.Position = UDim2.new(sat, 0, 1 - val, 0)
                HueKnob.Position = UDim2.new(hue, 0, 0.5, 0)
                if flag then Library.Flags[flag] = currentColor end
                if callback then task.spawn(callback, currentColor) end

            local svDrag, hueDrag = false, false
            local function svFromInput(input)
                local rx = math.clamp((input.Position.X - SV.AbsolutePosition.X) / SV.AbsoluteSize.X, 0, 1)
                local ry = math.clamp((input.Position.Y - SV.AbsolutePosition.Y) / SV.AbsoluteSize.Y, 0, 1)
                sat, val = rx, 1 - ry
                update()
            end
            local function hueFromInput(input)
                hue = math.clamp((input.Position.X - HueBar.AbsolutePosition.X) / HueBar.AbsoluteSize.X, 0, 1)
                update()
            end

                if i.UserInputType == Enum.UserInputType.MouseButton1
                or i.UserInputType == Enum.UserInputType.Touch then
                    svDrag = false
                    hueDrag = false
                end
            end))

            HexBox.FocusLost:Connect(function()
                local hex = HexBox.Text:gsub("#", "")
                local ok, col = pcall(Color3.fromHex, hex)
                if ok then
                    hue, sat, val = Color3.toHSV(col)
                    currentColor = col
                    update(true)
                else
                    HexBox.Text = "#" .. string.upper(currentColor:ToHex())
                end
            end)

            local openBtn = Create("TextButton", {
                Size = UDim2.fromOffset(90, 30),
                Position = UDim2.new(1, -90, 0.5, -15),
                BackgroundTransparency = 1,
                Text = "",

                end
            end)

                    hue, sat, val = Color3.toHSV(c)
                    currentColor = c
                    update(true)
                end,
                Get = function() return currentColor end,

                Tween(cs, 0.15, { Transparency = 0.4 })
            end)
            if clickCallback then
                chip.MouseButton1Click:Connect(function()
                    task.spawn(clickCallback)
                end)
            end
            return chip
        end

        --//////////////////////////////////////////////////////////////////////
        -- ЭЛЕМЕНТ: ImageLabel (картинка)
        --//////////////////////////////////////////////////////////////////////
        function Section:ImageLabel(assetId, height, config)
            config = config or {}
            local Container = Create("Frame", {
                Size = UDim2.new(1, 0, 0, height or 120),
                BackgroundColor3 = T.Bg1,
                BackgroundTransparency = 0.3,
                BorderSizePixel = 0,
                ClipsDescendants = true,
                Parent = Frame,
            })
            Corner(Container, 10)
            Stroke(Container, T.Border, 1, 0.5)

            local img = Create("ImageLabel", {
                Size = UDim2.new(1, -4, 1, -4),
                Position = UDim2.new(0, 2, 0, 2),
                BackgroundTransparency = 1,
                Image = type(assetId) == "number" and ("rbxassetid://" .. assetId) or tostring(assetId),
                ScaleType = config.ScaleType or Enum.ScaleType.Fit,
                Parent = Container,
            })
            Corner(img, 8)
            return {
                Set = function(_, v)
                    img.Image = type(v) == "number" and ("rbxassetid://" .. v) or tostring(v)
                end,
                Instance = Container,
            }
        end

        --//////////////////////////////////////////////////////////////////////
        -- РЕГИСТРИРУЕМ СЕКЦИЮ
        --//////////////////////////////////////////////////////////////////////
        table.insert(Tab.Sections, Section)
        return Section
    end

    return Tab
end

--==============================================================================
-- ПЕРЕОПРЕДЕЛЕНИЕ CreateWindow — добавляем метод Tab
--==============================================================================
local _origCreateWindow = Library.CreateWindow

function Library:CreateWindow(config)
    -- вызываем оригинальный CreateWindow из Part 1
    local Window = _origCreateWindow(self, config)

    -- НАВЕШИВАЕМ метод Tab (перебиваем метатаблицу прямым присваиванием)
    Window.Tab = function(_, name)
        local Tab = CreateTab(Window, name)
        table.insert(Window.Tabs, Tab)

        -- если это первая вкладка — выбираем её
        if not Window._getCurrentTab() then
            Window._env.SelectTab(Tab)
        end

        return Tab
    end

    return Window
end

--/////////////////////////////////////////////////////////////////////////////
-- Library:SaveConfig(name) — сохранить все Flags в файл
--/////////////////////////////////////////////////////////////////////////////
function Library:SaveConfig(name)
    if not writefile then
        return false, "writefile недоступен"
    end
    name = name or "rubz_config"
    local ok, json = pcall(function()
        return HttpService:JSONEncode(self.Flags)
    end)
    if not ok or not json then
        return false, "ошибка JSON-кодирования"
    end
    local ok2, err = pcall(writefile, name .. ".json", json)
    if not ok2 then
        return false, tostring(err)
    end
    self:Print("Конфиг сохранён:", name)
    return true
end

--/////////////////////////////////////////////////////////////////////////////
-- Library:LoadConfig(name) — загрузить Flags из файла
--/////////////////////////////////////////////////////////////////////////////
function Library:LoadConfig(name)
    if not (isfile and readfile) then
        return false, "readfile/isfile недоступны"
    end
    name = name or "rubz_config"
    if not isfile(name .. ".json") then
        return false, "файл не найден"
    end
    local ok, data = pcall(readfile, name .. ".json")
    if not ok or not data then
        return false, "не удалось прочитать файл"
    end
    local ok2, tbl = pcall(function()
        return HttpService:JSONDecode(data)
    end)
    if not ok2 or type(tbl) ~= "table" then
        return false, "ошибка JSON-декодирования"
    end
    for k, v in pairs(tbl) do
        self.Flags[k] = v
    end
    self:Print("Конфиг загружен:", name)
    return true
end

--/////////////////////////////////////////////////////////////////////////////
-- Library:DeleteConfig(name) — удалить файл конфига
--/////////////////////////////////////////////////////////////////////////////
function Library:DeleteConfig(name)
    if not delfile then return false, "delfile недоступен" end
    name = name or "rubz_config"
    local path = name .. ".json"
    if isfile and not isfile(path) then
        return false, "файл не найден"
    end
    pcall(delfile, path)
    return true
end

--/////////////////////////////////////////////////////////////////////////////
-- Library:ListConfigs() — список сохранённых конфигов
--/////////////////////////////////////////////////////////////////////////////
function Library:ListConfigs()
    if not listfiles then return {} end
    local files = listfiles(".")
    local configs = {}
    for _, f in ipairs(files) do
        local name = f:match("([^/\\]+)%.json$")
        if name then
            table.insert(configs, name)
        end
    end
    return configs
end

--/////////////////////////////////////////////////////////////////////////////
-- Library:SetAutoSave(Window, interval) — автосохранение флагов
--/////////////////////////////////////////////////////////////////////////////
function Library:SetAutoSave(interval)
    if not self:HasFileAPI() then
        self:Warn("Автосохранение недоступно — нет writefile")
        return nil
    end
    interval = interval or 30
    return task.spawn(function()
        while true do
            task.wait(interval)
            self:SaveConfig("rubz_autosave")
        end
    end)
end

-- Library:CreateToggleKey(Window, key, callback) — глобальная привязка клавиши
--/////////////////////////////////////////////////////////////////////////////
function Library:CreateToggleKey(key, callback)
    return UserInputService.InputBegan:Connect(function(input, gpe)
        if gpe then return end
        if input.KeyCode == key then
            if callback then task.spawn(callback) end
        end
    end)
end

    local char = LocalPlayer.Character

-- Library:GetRootPart() — получить HRP
--/////////////////////////////////////////////////////////////////////////////
function Library:GetRootPart()
    local char = LocalPlayer.Character
    if char then
        return char:FindFirstChild("HumanoidRootPart")
    end
    return nil
end

--/////////////////////////////////////////////////////////////////////////////
-- Library:OnCharacter(callback) — вызвать при спавне персонажа
--/////////////////////////////////////////////////////////////////////////////
function Library:OnCharacter(callback)
    if LocalPlayer.Character then
        task.spawn(callback, LocalPlayer.Character)
    end
    return LocalPlayer.CharacterAdded:Connect(function(char)
        task.wait(0.1)
        task.spawn(callback, char)
    end)
end

--/////////////////////////////////////////////////////////////////////////////
-- Library:OnDeath(callback) — вызвать при смерти персонажа
--/////////////////////////////////////////////////////////////////////////////
function Library:OnDeath(callback)
    local conn
    conn = LocalPlayer.CharacterAdded:Connect(function(char)
        local hum = char:WaitForChild("Humanoid", 5)
        if hum then
            hum.Died:Connect(function()
                task.spawn(callback)
            end)
        end
    end)
    return conn
end

--/////////////////////////////////////////////////////////////////////////////
-- Library:FindFirst(parent, className) — искать первый Instance по классу
--/////////////////////////////////////////////////////////////////////////////
function Library:FindFirst(parent, className)
    for _, v in ipairs(parent:GetDescendants()) do
        if v:IsA(className) then
            return v
        end
    end
    return nil
end

--/////////////////////////////////////////////////////////////////////////////
-- Library:FindAll(parent, className) — все Instance по классу
--/////////////////////////////////////////////////////////////////////////////
function Library:FindAll(parent, className)
    local list = {}
    for _, v in ipairs(parent:GetDescendants()) do
        if v:IsA(className) then
            table.insert(list, v)
        end
    end
    return list
end

--/////////////////////////////////////////////////////////////////////////////
-- Library:GetPlayers(filter) — список игроков, filter(plr) → bool
--/////////////////////////////////////////////////////////////////////////////
function Library:GetPlayers(filter)
    local list = {}
    for _, plr in ipairs(Players:GetPlayers()) do
        if plr ~= LocalPlayer then
            if not filter or filter(plr) then
                table.insert(list, plr)
            end
        end
    end
    return list
end

--/////////////////////////////////////////////////////////////////////////////
-- Library:GetNearestPlayer(maxDist) — ближайший игрок
--/////////////////////////////////////////////////////////////////////////////
function Library:GetNearestPlayer(maxDist)
    local myChar = LocalPlayer.Character
    local myRoot = myChar and myChar:FindFirstChild("HumanoidRootPart")
    if not myRoot then return nil end
    local nearest, nearestDist
    for _, plr in ipairs(Players:GetPlayers()) do
        if plr ~= LocalPlayer then
            local char = plr.Character
            local root = char and char:FindFirstChild("HumanoidRootPart")
            if root then
                local d = (root.Position - myRoot.Position).Magnitude
                if not maxDist or d <= maxDist then
                    if not nearestDist or d < nearestDist then
                        nearest = plr
                        nearestDist = d
                    end
                end
            end
        end
    end
    return nearest, nearestDist
end
