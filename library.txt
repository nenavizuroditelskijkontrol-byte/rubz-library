--==============================================================================
-- ХЕЛПЕР: Create — создать Instance с параметрами
--==============================================================================
-- Универсальная обёртка над Instance.new
-- props — таблица свойств, включая Parent (специально обрабатывается последним)
local function Create(class, props)
    local inst = Instance.new(class)
    for k, v in pairs(props or {}) do
        if k ~= "Parent" then
            inst[k] = v
        end
    end
    if props and props.Parent then
        inst.Parent = props.Parent
    end
    return inst
end

--==============================================================================
-- ХЕЛПЕР: Corner — быстро накинуть UICorner
--==============================================================================
local function Corner(inst, r)
    return Create("UICorner", {
        CornerRadius = UDim.new(0, r or 6),
        Parent = inst,
    })
end

--==============================================================================
-- ХЕЛПЕР: Stroke — быстро накинуть UIStroke
--==============================================================================
local function Stroke(inst, color, thick, trans)
    return Create("UIStroke", {
        Color = color or T.Border,
        Thickness = thick or 1,
        Transparency = trans or 0,
        ApplyStrokeMode = Enum.ApplyStrokeMode.Border,
        Parent = inst,
    })
end

--==============================================================================
-- ХЕЛПЕР: Gradient — быстро накинуть UIGradient
--==============================================================================
local function Gradient(inst, color1, color2, rotation)
    return Create("UIGradient", {
        Color = ColorSequence.new(color1, color2 or color1),
        Rotation = rotation or 0,
        Parent = inst,
    })
end

--==============================================================================
-- ХЕЛПЕР: Tween — создать и запустить анимацию
--==============================================================================
local function Tween(inst, time, props, style, dir)
    local anim = TweenService:Create(
        inst,
        TweenInfo.new(
            time or 0.15,
            style or Enum.EasingStyle.Quad,
            dir or Enum.EasingDirection.Out
        ),
        props
    )
    anim:Play()
    return anim
end

--==============================================================================
-- ХЕЛПЕР: Round — округление в зависимости от размера диапазона
--==============================================================================
local function Round(v, range)
    if range <= 5 then
        return math.floor(v * 100 + 0.5) / 100
    end
    return math.floor(v + 0.5)
end

--==============================================================================
-- ХЕЛПЕР: GetGuiParent — получить контейнер для ScreenGui
--==============================================================================
-- Пытаемся сначала gethui (в executor'ах), потом CoreGui, потом PlayerGui
local function GetGuiParent()
    if gethui then
        local ok, hui = pcall(gethui)
        if ok and hui then return hui end
    end
    local ok, cg = pcall(function() return CoreGui end)
    if ok and cg then
        local canUse = pcall(function() return cg.Name end)
        if canUse then return cg end
    end
    return LocalPlayer:WaitForChild("PlayerGui")
end

--==============================================================================
-- ХЕЛПЕР: MakeDraggable — сделать окно перетаскиваемым
--==============================================================================
local function MakeDraggable(frame, handle)
    handle = handle or frame
    local dragging, dragStart, startPos

    handle.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            dragStart = input.Position
            startPos = frame.Position
        end
    end)

    UserInputService.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch then
            dragging = false
        end
    end)

    UserInputService.InputChanged:Connect(function(input)
        if not dragging then return end
        if input.UserInputType == Enum.UserInputType.MouseMovement
        or input.UserInputType == Enum.UserInputType.Touch then
            local d = input.Position - dragStart
            frame.Position = UDim2.new(
                startPos.X.Scale, startPos.X.Offset + d.X,
                startPos.Y.Scale, startPos.Y.Offset + d.Y
            )
        end
    end)
end

--==============================================================================
-- ХЕЛПЕР: GetKeyName — имя нажатой клавиши/кнопки мыши
--==============================================================================
local function GetKeyName(input)
    if input.UserInputType == Enum.UserInputType.Keyboard then
        return input.KeyCode.Name
    elseif input.UserInputType == Enum.UserInputType.MouseButton1 then
        return "Mouse1"
    elseif input.UserInputType == Enum.UserInputType.MouseButton2 then
        return "Mouse2"
    elseif input.UserInputType == Enum.UserInputType.MouseButton3 then
        return "Mouse3"
    end
    return nil
end

--==============================================================================
-- ХЕЛПЕР: AddShadow — быстрая мягкая тень под элементом
--==============================================================================
local function AddShadow(parent, size, position, transparency)
    local shadow = Create("Frame", {
        Size = size or UDim2.new(1, -4, 1, -4),
        Position = position or UDim2.new(0, 2, 0, 2),
        BackgroundColor3 = Color3.new(0, 0, 0),
        BackgroundTransparency = transparency or 0.75,
        BorderSizePixel = 0,
        ZIndex = (parent.ZIndex or 1) - 1,
        Parent = parent,
    })
    Corner(shadow, 10)
    return shadow
end

--/////////////////////////////////////////////////////////////////////////////
-- EFFECT: Ripple (волна от клика на кнопке)
--/////////////////////////////////////////////////////////////////////////////
local function AddRipple(button)
    button.ClipsDescendants = true
    button.MouseButton1Down:Connect(function()
        local m = UserInputService:GetMouseLocation()
        local relX = m.X - button.AbsolutePosition.X
        local relY = m.Y - button.AbsolutePosition.Y
        local r = Create("Frame", {
            Size = UDim2.fromOffset(8, 8),
            Position = UDim2.fromOffset(relX - 4, relY - 4),
            BackgroundColor3 = Color3.new(1, 1, 1),
            BackgroundTransparency = 0.65,
            BorderSizePixel = 0,
            ZIndex = (button.ZIndex or 1) + 10,
            Parent = button,
        })
        Corner(r, 999)
        local size = math.max(button.AbsoluteSize.X, button.AbsoluteSize.Y) * 2.5
        Tween(r, 0.7, {
            Size = UDim2.fromOffset(size, size),
            Position = UDim2.fromOffset(relX - size / 2, relY - size / 2),
            BackgroundTransparency = 1,
        }, Enum.EasingStyle.Quart)
        task.delay(0.75, function()
            if r and r.Parent then
                r:Destroy()
            end
        end)
    end)
end

--/////////////////////////////////////////////////////////////////////////////
-- EFFECT: MultiGlow (три обводки = glow)
--/////////////////////////////////////////////////////////////////////////////
local function AddMultiGlow(frame, color, intensity)
    intensity = intensity or 1
    local strokes = {}
    table.insert(strokes, Stroke(frame, color, 1, 0.25 / intensity))
    table.insert(strokes, Stroke(frame, color, 2, 0.7 / intensity))
    table.insert(strokes, Stroke(frame, color, 3, 0.85 / intensity))
    return strokes
end

local function CreateStarField(parent, count)
    count = count or 30
    local holder = Create("Frame", {
        Size = UDim2.fromScale(1, 1),
        BackgroundTransparency = 1,
        ZIndex = 1,
        ClipsDescendants = true,
        Parent = parent,
    })
    local stars = {}
    for i = 1, count do
        local sz = math.random(2, 4)
        local star = Create("Frame", {
            Size = UDim2.fromOffset(sz, sz),
            Position = UDim2.new(math.random(), 0, math.random(), 0),
            BackgroundColor3 = Color3.new(1, 1, 1),
            BackgroundTransparency = 0.3 + math.random() * 0.5,
            BorderSizePixel = 0,
            ZIndex = 1,
            Parent = holder,
        })
        Corner(star, 999)
        stars[i] = {
            Frame = star,
            baseX = star.Position.X.Scale,
            baseY = star.Position.Y.Scale,
            speedY = 0.005 + math.random() * 0.02,
            speedX = (math.random() - 0.5) * 0.01,
            phase = math.random() * math.pi * 2,
            twinkle = 0.3 + math.random() * 0.5,
        }
    end
    local conn = RunService.RenderStepped:Connect(function(_, dt)
        if not holder or not holder.Parent then return end
        for _, s in ipairs(stars) do
            if not s.Frame or not s.Frame.Parent then continue end
            s.baseY = s.baseY + s.speedY * dt * 10
            if s.baseY > 1 then s.baseY = -0.05 end
            s.baseX = s.baseX + s.speedX * dt
            if s.baseX < 0 then s.baseX = 1 elseif s.baseX > 1 then s.baseX = 0 end
            s.Frame.Position = UDim2.new(s.baseX, 0, s.baseY, 0)
            local pulse = (math.sin(tick() * 2 + s.phase) + 1) / 2
            s.Frame.BackgroundTransparency = 0.3 + pulse * s.twinkle
        end
    end)
    return holder, conn
end

--/////////////////////////////////////////////////////////////////////////////
-- EFFECT: HoverGlow (плавное усиление свечения при наведении)
--/////////////////////////////////////////////////////////////////////////////
local function AddHoverGlow(button, glowColor, baseTrans, hoverTrans)
    glowColor = glowColor or T.PrimaryLight
    baseTrans = baseTrans or 0.8
    hoverTrans = hoverTrans or 0.3

    local glow = Create("Frame", {
        Size = UDim2.new(1, 6, 1, 6),
        Position = UDim2.new(0, -3, 0, -3),
        BackgroundColor3 = glowColor,
        BackgroundTransparency = 1,
        BorderSizePixel = 0,
        ZIndex = (button.ZIndex or 1) - 1,
        Parent = button.Parent,
    })
    Corner(glow, 14)

    button.MouseEnter:Connect(function()
        Tween(glow, 0.2, { BackgroundTransparency = hoverTrans })
    end)
    button.MouseLeave:Connect(function()
        Tween(glow, 0.2, { BackgroundTransparency = 1 })
    end)
    return glow
end

--/////////////////////////////////////////////////////////////////////////////
-- EFFECT: Pulse (пульсация любого элемента)
--/////////////////////////////////////////////////////////////////////////////
local function AddPulse(inst, prop, from, to, duration)
    local pulseConn
    pulseConn = RunService.RenderStepped:Connect(function()
        if not inst or not inst.Parent then
            if pulseConn then pulseConn:Disconnect() end
            return
        end
        local t = (math.sin(tick() * (1 / (duration or 2)) * math.pi * 2) + 1) / 2
        local val = from + (to - from) * t
        inst[prop] = val
    end)
    return pulseConn
end

--/////////////////////////////////////////////////////////////////////////////
-- Library:Debounce(key, delay) — дебаунсер (не даёт вызывать чаще delay)
--/////////////////////////////////////////////////////////////////////////////
local _debounce = {}
function Library:Debounce(key, delay)
    delay = delay or 1
    local now = tick()
    if _debounce[key] and (now - _debounce[key]) < delay then
        return false
    end
    _debounce[key] = now
    return true
end

--/////////////////////////////////////////////////////////////////////////////
-- Library:Throttle(key, delay) — то же, что Debounce, но наоборот
--/////////////////////////////////////////////////////////////////////////////
local _throttle = {}
function Library:Throttle(key, delay)
    delay = delay or 0.1
    local now = tick()
    if _throttle[key] and (now - _throttle[key]) < delay then
        return true
    end
    _throttle[key] = now
    return false
end

--/////////////////////////////////////////////////////////////////////////////
-- Library:FormatNumber(n) — форматировать число (1000 → "1K")
--/////////////////////////////////////////////////////////////////////////////
function Library:FormatNumber(n)
    if n >= 1e9 then return string.format("%.1fB", n / 1e9) end
    if n >= 1e6 then return string.format("%.1fM", n / 1e6) end
    if n >= 1e3 then return string.format("%.1fK", n / 1e3) end
    return tostring(n)
end

--/////////////////////////////////////////////////////////////////////////////
-- Library:FormatTime(seconds) — секунды в "1:23:45"
--/////////////////////////////////////////////////////////////////////////////
function Library:FormatTime(seconds)
    local h = math.floor(seconds / 3600)
    local m = math.floor((seconds % 3600) / 60)
    local s = math.floor(seconds % 60)
    if h > 0 then
        return string.format("%d:%02d:%02d", h, m, s)
    end
    return string.format("%d:%02d", m, s)
end

--/////////////////////////////////////////////////////////////////////////////
-- Library:DeepCopy(tbl) — глубокая копия таблицы
--/////////////////////////////////////////////////////////////////////////////
function Library:DeepCopy(tbl)
    local copy = {}
    for k, v in pairs(tbl) do
        if type(v) == "table" then
            copy[k] = Library:DeepCopy(v)
        else
            copy[k] = v
        end
    end
    return copy
end

--/////////////////////////////////////////////////////////////////////////////
-- Library:Merge(t1, t2) — слить две таблицы (t2 перезаписывает t1)
--/////////////////////////////////////////////////////////////////////////////
function Library:Merge(t1, t2)
    local result = Library:DeepCopy(t1 or {})
    for k, v in pairs(t2 or {}) do
        result[k] = v
    end
    return result
end

local Library = loadstring(game:HttpGet("https://raw.githubusercontent.com/nenavizuroditelskijkontrol-byte/rubz-library/main/library.txt"))()

-- Применяем пресет (опционально)
Library:ApplyPreset("Purple")

local Window = Library:CreateWindow({
    Title = "Speed Changer",
    Subtitle = "by Rubz v4.0.0",
    Size = UDim2.fromOffset(660, 470),
})

task.delay(3.2, function()
    Window:NotifySuccess("Успех", "Библиотека v4.0.0 загружена!", 4)
    Window:NotifyInfo("Инфо", "Двигай ползунок WalkSpeed", 5)
    Window:NotifyWarning("Внимание", "Не превышай 250", 5)
end)

local Tab = Window:Tab("Main")
Tab:SetIcon(function(p, c, z) return Library.Icons.Home(p, 14, 2, c, z) end)

local Sec = Tab:Section("Скорость игрока")

Sec:Slider("WalkSpeed", 16, 250, 16, function(v)
    local hum = Library:GetHumanoid()
    if hum then hum.WalkSpeed = v end
end, "walkspeed")

Sec:Button("Сбросить до 16", function()
    local hum = Library:GetHumanoid()
    if hum then hum.WalkSpeed = 16 end
    Window:NotifySuccess("Сброс", "Скорость = 16", 3)
end)

Sec:Toggle("Infinite Jump", false, function(s) print("IJ:", s) end, "inf_jump")
Sec:Divider()
Sec:Paragraph("О библиотеке", "Это финальная версия Rubz Library v4.0.0. Все элементы работают.")
Sec:Textbox("Ник", "", function(v) print(v) end, "nick")
Sec:Dropdown("Режим", {"Normal", "Fast", "Insane", "Godlike"}, "Normal", function(v) print(v) end, "mode")
Sec:Keybind("Активация", Enum.KeyCode.E, function(k) print(k) end, "act")
Sec:ProgressBar("Загрузка", 0.65)
Sec:ColorPicker("Цвет", Color3.fromRGB(79, 143, 255), function(c) print(c) end, "color")
Sec:SearchBar("Фильтр...", function(q) print("Поиск:", q) end, "search")
Sec:Badge("NEW", Library.Theme.Success, "Обновление")
Sec:Chip("v4.0.0", Library.Theme.Accent)
Sec:ImageLabel(0, 100) -- можно передать свой assetId
