local function Create(class, props)
    local inst = Instance.new(class)
    for k, v in pairs(props or {}) do
        if k ~= "Parent" then inst[k] = v end
    end
    if props and props.Parent then inst.Parent = props.Parent end
    return inst
end

local function Corner(inst, r)
    return Create("UICorner", { CornerRadius = UDim.new(0, r or 6), Parent = inst })
end

local function Stroke(inst, color, thick, trans)
    return Create("UIStroke", {
        Color = color or Theme.Outline,
        Thickness = thick or 1,
        Transparency = trans or 0,
        ApplyStrokeMode = Enum.ApplyStrokeMode.Border,
        Parent = inst,
    })
end

local function Grad(inst, c1, c2, rot)
    return Create("UIGradient", {
        Color = ColorSequence.new(c1, c2),
        Rotation = rot or 0,
        Parent = inst,
    })
end

local function Tween(inst, t, props, style, dir)
    local anim = TweenService:Create(
        inst,
        TweenInfo.new(t or 0.15, style or Enum.EasingStyle.Quad, dir or Enum.EasingDirection.Out),
        props
    )
    anim:Play()
    return anim
end

local function Round(v, range)
    if range <= 5 then return math.floor(v * 100 + 0.5) / 100 end
    return math.floor(v + 0.5)
end

local function GetGuiParent()
    if gethui then
        local ok, hui = pcall(gethui)
        if ok and hui then return hui end
    end
    local ok, cg = pcall(function() return CoreGui end)
    if ok and cg then
        local canUse = pcall(function() return cg.Name end)
        if canUse then return cg end
    end
    return LocalPlayer:WaitForChild("PlayerGui")
end

local function MakeDraggable(frame, handle)
    handle = handle or frame
    local dragging, dragStart, startPos

    handle.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            dragStart = input.Position
            startPos = frame.Position
        end
    end)
    UserInputService.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch then
            dragging = false
        end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if not dragging then return end
        if input.UserInputType == Enum.UserInputType.MouseMovement
        or input.UserInputType == Enum.UserInputType.Touch then
            local delta = input.Position - dragStart
            frame.Position = UDim2.new(
                startPos.X.Scale, startPos.X.Offset + delta.X,
                startPos.Y.Scale, startPos.Y.Offset + delta.Y
            )
        end
    end)
end

local function GetKeyName(input)
    if input.UserInputType == Enum.UserInputType.Keyboard then
        return input.KeyCode.Name
    elseif input.UserInputType == Enum.UserInputType.MouseButton1 then return "Mouse1"
    elseif input.UserInputType == Enum.UserInputType.MouseButton2 then return "Mouse2"
    elseif input.UserInputType == Enum.UserInputType.MouseButton3 then return "Mouse3"
    end
end

-- Иконки рисуем сами (шрифты робят непредсказуемо)
local function DrawX(parent, size, thickness, color, zIndex)
    size = size or 12
    thickness = thickness or 2
    local container = Create("Frame", {
        Size = UDim2.fromOffset(size, size),
        AnchorPoint = Vector2.new(0.5, 0.5),
        Position = UDim2.new(0.5, 0, 0.5, 0),
        BackgroundTransparency = 1,
        ZIndex = zIndex or 2,
        Parent = parent,
    })
    local l1 = Create("Frame", {
        Size = UDim2.fromOffset(size, thickness),
        AnchorPoint = Vector2.new(0.5, 0.5),
        Position = UDim2.new(0.5, 0, 0.5, 0),
        BackgroundColor3 = color or Theme.Text,
        BorderSizePixel = 0,
        Rotation = 45,
        ZIndex = zIndex or 2,
        Parent = container,
    })
    Corner(l1, thickness / 2)
    local l2 = Create("Frame", {
        Size = UDim2.fromOffset(size, thickness),
        AnchorPoint = Vector2.new(0.5, 0.5),
        Position = UDim2.new(0.5, 0, 0.5, 0),
        BackgroundColor3 = color or Theme.Text,
        BorderSizePixel = 0,
        Rotation = -45,
        ZIndex = zIndex or 2,
        Parent = container,
    })
    Corner(l2, thickness / 2)
    return container, { l1, l2 }
end

local function DrawMinus(parent, width, thickness, color, zIndex)
    local line = Create("Frame", {
        Size = UDim2.fromOffset(width or 12, thickness or 2),
        AnchorPoint = Vector2.new(0.5, 0.5),
        Position = UDim2.new(0.5, 0, 0.5, 0),
        BackgroundColor3 = color or Theme.Text,
        BorderSizePixel = 0,
        ZIndex = zIndex or 2,
        Parent = parent,
    })
    Corner(line, (thickness or 2) / 2)
    return line
end

local function DrawChevron(parent, size, thickness, color, zIndex)
    local container = Create("Frame", {
        Size = UDim2.fromOffset(size or 10, size or 10),
        AnchorPoint = Vector2.new(0.5, 0.5),
        Position = UDim2.new(0.5, 0, 0.5, 0),
        BackgroundTransparency = 1,
        ZIndex = zIndex or 2,
        Parent = parent,
    })
    local a = Create("Frame", {
        Size = UDim2.fromOffset(size or 10, thickness or 2),
        AnchorPoint = Vector2.new(0.5, 0.5),
        Position = UDim2.new(0.5, -3, 0.5, -1),
        BackgroundColor3 = color or Theme.Text,
        BorderSizePixel = 0,
        Rotation = 45,
        ZIndex = zIndex or 2,
        Parent = container,
    })
    Corner(a, 1)
    local b = Create("Frame", {
        Size = UDim2.fromOffset(size or 10, thickness or 2),
        AnchorPoint = Vector2.new(0.5, 0.5),
        Position = UDim2.new(0.5, 3, 0.5, -1),
        BackgroundColor3 = color or Theme.Text,
        BorderSizePixel = 0,
        Rotation = -45,
        ZIndex = zIndex or 2,
        Parent = container,
    })
    Corner(b, 1)
    return container, { a, b }
end

-- Ripple
local function AddRipple(button)
    button.ClipsDescendants = true
    button.MouseButton1Down:Connect(function()
        local mPos = UserInputService:GetMouseLocation()
        local relX = mPos.X - button.AbsolutePosition.X
        local relY = mPos.Y - button.AbsolutePosition.Y
        local ripple = Create("Frame", {
            Size = UDim2.fromOffset(10, 10),
            Position = UDim2.fromOffset(relX - 5, relY - 5),
            BackgroundColor3 = Color3.fromRGB(255, 255, 255),
            BackgroundTransparency = 0.7,
            BorderSizePixel = 0,
            ZIndex = (button.ZIndex or 1) + 5,
            Parent = button,
        })
        Corner(ripple, 999)
        local size = math.max(button.AbsoluteSize.X, button.AbsoluteSize.Y) * 2.3
        Tween(ripple, 0.6, {
            Size = UDim2.fromOffset(size, size),
            Position = UDim2.fromOffset(relX - size / 2, relY - size / 2),
            BackgroundTransparency = 1,
        }, Enum.EasingStyle.Quart)
        task.delay(0.65, function()
            if ripple and ripple.Parent then ripple:Destroy() end
        end)
    end)
end

-- Пульсирующий glow вокруг элемента (используется для акцентных штук)
local function AddGlow(frame, color, thickness, trans)
    local g = Stroke(frame, color or Theme.AccentGlow, thickness or 1.5, trans or 0.4)
    return g
end

function Library:Intro(config)
    config = config or {}
    local duration = config.Duration or 2.8
    local title    = config.Title or "Rubz Library"
    local subtitle = config.Subtitle or ("v" .. Library.Version)
    local parent   = config.Parent or GetGuiParent()
    local ownGui   = not config.Gui

    local gui = config.Gui or Create("ScreenGui", {
        Name = "RubzIntro_" .. tostring(math.random(1, 1e6)),
        ResetOnSpawn = false,
        IgnoreGuiInset = true,
        ZIndexBehavior = Enum.ZIndexBehavior.Sibling,
        DisplayOrder = 999,
        Parent = parent,
    })

    local Backdrop = Create("Frame", {
        Size = UDim2.new(1, 0, 1, 0),
        BackgroundColor3 = Color3.fromRGB(0, 0, 0),
        BackgroundTransparency = 1,
        BorderSizePixel = 0,
        Parent = gui,
    })

    local Card = Create("Frame", {
        Size = UDim2.fromOffset(0, 0),
        AnchorPoint = Vector2.new(0.5, 0.5),
        Position = UDim2.new(0.5, 0, 0.5, 0),
        BackgroundColor3 = Color3.fromRGB(255, 255, 255), -- важно: даём цвет градиенту
        BorderSizePixel = 0,
        ClipsDescendants = true,
        Parent = gui,
    })
    Corner(Card, 16)
    Stroke(Card, Theme.Accent, 1.5, 0.3)
    Stroke(Card, Theme.AccentBright, 3, 0.85)

    local CardGrad = Create("UIGradient", {
        Color = ColorSequence.new({
            ColorSequenceKeypoint.new(0.0, Color3.fromRGB(8, 12, 26)),
            ColorSequenceKeypoint.new(0.4, Color3.fromRGB(30, 70, 170)),
            ColorSequenceKeypoint.new(0.5, Color3.fromRGB(120, 180, 255)),
            ColorSequenceKeypoint.new(0.6, Color3.fromRGB(30, 70, 170)),
            ColorSequenceKeypoint.new(1.0, Color3.fromRGB(8, 12, 26)),
        }),
        Rotation = 0,
        Parent = Card,
    })

    local Title = Create("TextLabel", {
        Size = UDim2.new(1, 0, 0, 32),
        Position = UDim2.new(0, 0, 0.5, -30),
        BackgroundTransparency = 1,
        Text = title,
        TextColor3 = Color3.new(1, 1, 1),
        TextSize = 26,
        Font = Enum.Font.GothamBold,
        TextTransparency = 1,
        Parent = Card,
    })
    local Sub = Create("TextLabel", {
        Size = UDim2.new(1, 0, 0, 16),
        Position = UDim2.new(0, 0, 0.5, 4),
        BackgroundTransparency = 1,
        Text = subtitle,
        TextColor3 = Color3.fromRGB(210, 225, 255),
        TextSize = 13,
        Font = Enum.Font.Gotham,
        TextTransparency = 1,
        Parent = Card,
    })

    local BarBg = Create("Frame", {
        Size = UDim2.new(0, 240, 0, 4),
        Position = UDim2.new(0.5, -120, 1, -28),
        BackgroundColor3 = Color3.fromRGB(255, 255, 255),
        BackgroundTransparency = 0.75,
        BorderSizePixel = 0,
        Parent = Card,
    })
    Corner(BarBg, 2)

    local BarFill = Create("Frame", {
        Size = UDim2.new(0, 0, 1, 0),
        BackgroundColor3 = Color3.fromRGB(255, 255, 255),
        BorderSizePixel = 0,
        Parent = BarBg,
    })
    Corner(BarFill, 2)
    Create("UIGradient", {
        Color = ColorSequence.new({
            ColorSequenceKeypoint.new(0, Color3.fromRGB(150, 200, 255)),
            ColorSequenceKeypoint.new(1, Color3.new(1, 1, 1)),
        }),
        Parent = BarFill,
    })

    Tween(Backdrop, 0.25, { BackgroundTransparency = 0.4 })
    Tween(Card, 0.45, { Size = UDim2.fromOffset(320, 150) }, Enum.EasingStyle.Back)
    task.delay(0.15, function()
        Tween(Title, 0.3, { TextTransparency = 0 })
        Tween(Sub, 0.3, { TextTransparency = 0 })
    end)

    local spin = RunService.RenderStepped:Connect(function(_, dt)
        if CardGrad and CardGrad.Parent then
            CardGrad.Rotation = (CardGrad.Rotation + dt * 70) % 360
        end
    end)

    Tween(BarFill, duration * 0.85, { Size = UDim2.new(1, 0, 1, 0) }, Enum.EasingStyle.Quad)

    task.delay(duration, function()
        if spin then spin:Disconnect() end
        if Card and Card.Parent then
            Tween(Card, 0.3, { Size = UDim2.fromOffset(0, 0) }, Enum.EasingStyle.Back, Enum.EasingDirection.In)
            Tween(Backdrop, 0.35, { BackgroundTransparency = 1 })
            task.delay(0.4, function()
                if ownGui then gui:Destroy() end
            end)
        end
    end)

    return gui
end

function Library:CreateWindow(config)
    config = config or {}
    local Title     = config.Title or "Rubz UI"
    local Subtitle  = config.Subtitle or ("v" .. Library.Version)
    local Size      = config.Size or UDim2.fromOffset(620, 440)
    local ShowIntro = config.Intro ~= false

    if ShowIntro then
        Library:Intro({
            Title = config.IntroTitle or "Rubz Library",
            Subtitle = config.IntroSubtitle or ("v" .. Library.Version),
            Duration = config.IntroDuration or 2.8,
        })
    end

    local Window = {}
    Window.Tabs = {}
    Window.Flags = Library.Flags
    Window._conns = {}
    Window._activeTab = nil

    local ScreenGui = Create("ScreenGui", {
        Name = "RubzUI_" .. tostring(math.random(1, 1e6)),
        ResetOnSpawn = false,
        ZIndexBehavior = Enum.ZIndexBehavior.Sibling,
        IgnoreGuiInset = true,
        Parent = GetGuiParent(),
    })

    --//////////////////////////////////////////////////////////////////////////
    -- AMBIENT GLOW — размытое свечение вокруг окна
    --//////////////////////////////////////////////////////////////////////////
    local GlowLayer = Create("Frame", {
        Name = "GlowLayer",
        Size = Size + UDim2.fromOffset(60, 60),
        Position = UDim2.new(0.5, -(Size.X.Offset + 60) / 2, 0.5, -(Size.Y.Offset + 60) / 2),
        BackgroundColor3 = Theme.Accent,
        BackgroundTransparency = 1,
        BorderSizePixel = 0,
        ZIndex = 0,
        Parent = ScreenGui,
    })
    Corner(GlowLayer, 30)
    local glowConn = RunService.RenderStepped:Connect(function()
        if not GlowLayer or not GlowLayer.Parent then return end
        local pulse = (math.sin(tick() * 2) + 1) / 2
        GlowLayer.BackgroundTransparency = 0.88 + pulse * 0.06
    end)
    table.insert(Window._conns, glowConn)

    --//////////////////////////////////////////////////////////////////////////
    -- MAIN
    --//////////////////////////////////////////////////////////////////////////
    local Main = Create("Frame", {
        Name = "Main",
        Size = UDim2.fromOffset(0, 0),
        Position = UDim2.new(0.5, 0, 0.5, 0),
        AnchorPoint = Vector2.new(0.5, 0.5),
        -- ВАЖНО: белый фон, чтобы градиент был виден
        BackgroundColor3 = Color3.new(1, 1, 1),
        BorderSizePixel = 0,
        ClipsDescendants = true,
        ZIndex = 5,
        Parent = ScreenGui,
    })
    Corner(Main, 14)
    -- многослойная обводка = эффект свечения
    Stroke(Main, Theme.AccentBright, 1, 0.55)
    Stroke(Main, Theme.Accent, 1.5, 0.75)

    Tween(Main, 0.45, { Size = Size }, Enum.EasingStyle.Back)
    Tween(GlowLayer, 0.45, {
        Size = Size + UDim2.fromOffset(60, 60),
        Position = UDim2.new(0.5, -(Size.X.Offset + 60) / 2, 0.5, -(Size.Y.Offset + 60) / 2),
    }, Enum.EasingStyle.Back)

    --//////////////////////////////////////////////////////////////////////////
    -- АНИМИРОВАННЫЙ ГРАДИЕНТ (многослойный)
    --//////////////////////////////////////////////////////////////////////////
    local BgBase = Create("UIGradient", {
        Color = ColorSequence.new({
            ColorSequenceKeypoint.new(0.00, Color3.fromRGB(10, 14, 28)),
            ColorSequenceKeypoint.new(0.30, Color3.fromRGB(30, 65, 155)),
            ColorSequenceKeypoint.new(0.50, Color3.fromRGB(90, 160, 255)),
            ColorSequenceKeypoint.new(0.70, Color3.fromRGB(30, 65, 155)),
            ColorSequenceKeypoint.new(1.00, Color3.fromRGB(10, 14, 28)),
        }),
        Rotation = 20,
        Parent = Main,
    })
    local BgShine = Create("UIGradient", {
        Color = ColorSequence.new({
            ColorSequenceKeypoint.new(0.0, Color3.fromRGB(40, 60, 100)),
            ColorSequenceKeypoint.new(0.5, Color3.fromRGB(200, 230, 255)),
            ColorSequenceKeypoint.new(1.0, Color3.fromRGB(40, 60, 100)),
        }),
        Transparency = NumberSequence.new({
            NumberSequenceKeypoint.new(0.0, 1),
            NumberSequenceKeypoint.new(0.5, 0.6),
            NumberSequenceKeypoint.new(1.0, 1),
        }),
        Rotation = 135,
        Parent = Main,
    })
    local BgPulse = Create("UIGradient", {
        Color = ColorSequence.new({
            ColorSequenceKeypoint.new(0.0, Color3.fromRGB(60, 120, 255)),
            ColorSequenceKeypoint.new(0.5, Color3.fromRGB(180, 220, 255)),
            ColorSequenceKeypoint.new(1.0, Color3.fromRGB(60, 120, 255)),
        }),
        Transparency = NumberSequence.new({
            NumberSequenceKeypoint.new(0.0, 0.9),
            NumberSequenceKeypoint.new(0.5, 0.7),
            NumberSequenceKeypoint.new(1.0, 0.9),
        }),
        Rotation = 300,
        Parent = Main,
    })

    -- декоративные орбы (плавающие круги)
    local Orbs = {}
    local orbColors = { Theme.Accent, Theme.Purple, Theme.Pink, Theme.Cyan }
    for i = 1, 5 do
        local sz = math.random(100, 180)
        local Orb = Create("Frame", {
            Size = UDim2.fromOffset(sz, sz),
            Position = UDim2.new(math.random(), -sz / 2, math.random(), -sz / 2),
            BackgroundColor3 = orbColors[i],
            BackgroundTransparency = 0.85,
            BorderSizePixel = 0,
            ZIndex = 6,
            Parent = Main,
        })
        Corner(Orb, 999)
        Create("UIGradient", {
            Color = ColorSequence.new({
                ColorSequenceKeypoint.new(0, Color3.new(1, 1, 1)),
                ColorSequenceKeypoint.new(1, orbColors[i]),
            }),
            Transparency = NumberSequence.new({
                NumberSequenceKeypoint.new(0, 0.5),
                NumberSequenceKeypoint.new(1, 1),
            }),
            Rotation = math.random(0, 360),
            Parent = Orb,
        })
        Orbs[i] = {
            Frame = Orb,
            speed = 0.06 + math.random() * 0.12,
            phase = math.random() * math.pi * 2,
            baseX = math.random(),
            baseY = math.random(),
        }
    end

    --//////////////////////////////////////////////////////////////////////////
    -- АНИМАТОР ГРАДИЕНТА
    --//////////////////////////////////////////////////////////////////////////
    local gradT = 0
    local gradConn = RunService.RenderStepped:Connect(function(_, dt)
        if not Main or not Main.Parent then return end
        gradT = (gradT + dt * 0.4) % 1

        if BgBase.Parent then
            BgBase.Rotation = 20 + math.sin(gradT * math.pi * 2) * 25
        end
        if BgShine.Parent then
            BgShine.Offset = Vector2.new(math.sin(gradT * math.pi * 2) * 0.6, 0)
        end
        if BgPulse.Parent then
            BgPulse.Rotation = (gradT * 360 + 300) % 360
            BgPulse.Offset = Vector2.new(math.cos(gradT * math.pi * 3) * 0.25, 0)
        end
        for _, o in ipairs(Orbs) do
            if o.Frame and o.Frame.Parent then
                local x = o.baseX + math.sin(gradT * math.pi * 2 * o.speed + o.phase) * 0.18
                local y = o.baseY + math.cos(gradT * math.pi * 2 * o.speed + o.phase) * 0.18
                o.Frame.Position = UDim2.new(x, -o.Frame.AbsoluteSize.X / 2, y, -o.Frame.AbsoluteSize.Y / 2)
            end
        end
    end)
    table.insert(Window._conns, gradConn)

    --//////////////////////////////////////////////////////////////////////////
    -- TOPBAR (glass)
    --//////////////////////////////////////////////////////////////////////////
    local Topbar = Create("Frame", {
        Name = "Topbar",
        Size = UDim2.new(1, 0, 0, 48),
        BackgroundColor3 = Theme.Panel,
        BackgroundTransparency = Library.Opacity.Topbar,
        BorderSizePixel = 0,
        ClipsDescendants = true,
        ZIndex = 20,
        Parent = Main,
    })
    Corner(Topbar, 14)
    -- закрываем нижние углы
    Create("Frame", {
        Size = UDim2.new(1, 0, 0, 14),
        Position = UDim2.new(0, 0, 1, -14),
        BackgroundColor3 = Theme.Panel,
        BackgroundTransparency = Library.Opacity.Topbar,
        BorderSizePixel = 0,
        ZIndex = 20,
        Parent = Topbar,
    })

    -- лёгкий верхний градиент
    Create("UIGradient", {
        Color = ColorSequence.new({
            ColorSequenceKeypoint.new(0, Color3.fromRGB(255, 255, 255)),
            ColorSequenceKeypoint.new(1, Color3.fromRGB(180, 210, 255)),
        }),
        Transparency = NumberSequence.new({
            NumberSequenceKeypoint.new(0, 0.9),
            NumberSequenceKeypoint.new(1, 1),
        }),
        Rotation = 90,

    local currentTab
    local function SelectTab(tab)
        if currentTab == tab then return end
        if currentTab and currentTab.Page then
            currentTab.Page.Visible = false
        end
        for _, t in ipairs(Window.Tabs) do
            if t == tab then
                t.Page.Visible = true
                t.Page.GroupTransparency = 1
                Tween(t.Page, 0.25, { GroupTransparency = 0 })
                Tween(t.Button, 0.2, {
                    BackgroundTransparency = 0.35,
                    BackgroundColor3 = Theme.Element,
                })
                Tween(t.ButtonStroke, 0.2, { Color = Theme.AccentBright, Transparency = 0.4 })
                Tween(t.TextLabel, 0.2, { TextColor3 = Theme.Text })
                Tween(t.Indicator, 0.25, {
                    BackgroundTransparency = 0,
                    Size = UDim2.new(0, 3, 0, 22),
                }, Enum.EasingStyle.Back)
            else
                if t.Page.Visible then t.Page.Visible = false end
                Tween(t.Button, 0.2, { BackgroundTransparency = 1 })
                Tween(t.ButtonStroke, 0.2, { Color = Theme.OutlineSoft, Transparency = 1 })
                Tween(t.TextLabel, 0.2, { TextColor3 = Theme.SubText })
                Tween(t.Indicator, 0.25, {
                    BackgroundTransparency = 1,
                    Size = UDim2.new(0, 3, 0, 12),
                }, Enum.EasingStyle.Quart)
            end
        end
        currentTab = tab
        Window._activeTab = tab
    end

                    state = value and true or false
                    Tween(Switch, 0.22, {
                        BackgroundColor3 = state and Theme.Accent or Theme.Element,
                        BackgroundTransparency = state and 0 or Library.Opacity.Element,
                    }, Enum.EasingStyle.Quart)
                    Tween(Knob, 0.28, {
                        Position = state and UDim2.new(1, -19, 0.5, -8) or UDim2.new(0, 3, 0.5, -8),
                    }, Enum.EasingStyle.Back)
                    Tween(SwitchStroke, 0.2, {
                        Color = state and Theme.AccentBright or Theme.OutlineSoft,
                        Transparency = state and 0.25 or 0.6,
                    })
                    if flag then Library.Flags[flag] = state end
                    if not silent and callback then task.spawn(callback, state) end
                end
                Hit.MouseButton1Click:Connect(function() set(not state) end)
                if flag then Library.Flags[flag] = state end

                return {
                    Set = function(_, v) set(v, true) end,
                    Get = function() return state end,
                    Instance = Container,
                }
            end

                    BackgroundTransparency = 1,
                    Text = "",
                    Parent = Container,

                local function apply(alpha, silent)
                    alpha = math.clamp(alpha, 0, 1)
                    value = Round(min + range * alpha, range)
                    Fill.Size = UDim2.new(alpha, 0, 1, 0)
                    Knob.Position = UDim2.new(alpha, 0, 0.5, 0)
                    ValueLabel.Text = tostring(value)
                    if flag then Library.Flags[flag] = value end
                    if not silent and callback then task.spawn(callback, value) end
                end

                    local rel = (input.Position.X - Bar.AbsolutePosition.X) / Bar.AbsoluteSize.X
                    apply(rel)

                    if input.UserInputType == Enum.UserInputType.MouseButton1
                    or input.UserInputType == Enum.UserInputType.Touch then

                    end
                end)

                    if flag then Library.Flags[flag] = value end
                end

                    Set = function(_, v)
                        local alpha = range > 0 and (v - min) / range or 0
                        apply(alpha, true)
                    end,
                    Get = function() return value end,
                    Instance = Container,
                }

            function Section:Keybind(text, default, callback, flag)
                local currentKey = default or "None"
                local listening = false
                local Container = Create("Frame", {
                    Size = UDim2.new(1, 0, 0, 34),
                    BackgroundTransparency = 1,
                    Parent = Frame,
                })
                Create("TextLabel", {
                    Size = UDim2.new(1, -90, 1, 0),
                    BackgroundTransparency = 1,
                    Text = text,
                    TextColor3 = Theme.Text,
                    TextSize = 14,
                    Font = Enum.Font.Gotham,
                    TextXAlignment = Enum.TextXAlignment.Left,
                    Parent = Container,
                })
                local KeyBtn = Create("TextButton", {
                    Size = UDim2.fromOffset(80, 28),
                    Position = UDim2.new(1, -80, 0.5, -14),
                    BackgroundColor3 = Theme.Element,
                    BackgroundTransparency = Library.Opacity.Element,
                    Text = tostring(currentKey),
                    TextColor3 = Theme.Text,
                    TextSize = 13,
                    Font = Enum.Font.GothamMedium,
                    AutoButtonColor = false,
                    BorderSizePixel = 0,
                    Parent = Container,
                })
                Corner(KeyBtn, 8)
                local KBStroke = Stroke(KeyBtn, Theme.OutlineSoft, 1, 0.6)
                if flag then Library.Flags[flag] = currentKey end

                KeyBtn.MouseButton1Click:Connect(function()
                    if listening then return end
                    listening = true
                    KeyBtn.Text = "..."
                    Tween(KeyBtn, 0.15, { BackgroundColor3 = Theme.Accent, BackgroundTransparency = 0.2 })
                    Tween(KBStroke, 0.15, { Color = Theme.AccentBright, Transparency = 0 })
                end)

                local conn = UserInputService.InputBegan:Connect(function(input, gpe)
                    if listening then
                        if input.UserInputType == Enum.UserInputType.Keyboard
                        or input.UserInputType == Enum.UserInputType.MouseButton1
                        or input.UserInputType == Enum.UserInputType.MouseButton2
                        or input.UserInputType == Enum.UserInputType.MouseButton3 then
                            local name = GetKeyName(input) or "None"
                            if name == "Escape" then name = "None" end
                            currentKey = name
                            KeyBtn.Text = name
                            listening = false
                            Tween(KeyBtn, 0.15, { BackgroundColor3 = Theme.Element, BackgroundTransparency = Library.Opacity.Element })
                            Tween(KBStroke, 0.15, { Color = Theme.OutlineSoft, Transparency = 0.6 })
                            if flag then Library.Flags[flag] = currentKey end
                            if callback then task.spawn(callback, currentKey) end
                        end
                        return
                    end
                    if gpe then return end
                    if currentKey == "None" then return end
                    local name = GetKeyName(input)
                    if name and name == currentKey then
                        if callback then task.spawn(callback, currentKey) end
                    end
                end)
                table.insert(Window._conns, conn)

                return {
                    Set = function(_, v)
                        currentKey = v
                        KeyBtn.Text = tostring(v)
                        if flag then Library.Flags[flag] = v end
                    end,
                    Get = function() return currentKey end,
                    Instance = Container,
                }
            end

                        hueDrag = false

                HexBox.FocusLost:Connect(function()
                    local hex = HexBox.Text:gsub("#", "")
                    local ok, col = pcall(Color3.fromHex, hex)
                    if ok then
                        hue, sat, val = Color3.toHSV(col)
                        currentColor = col
                        updateAll(true)
                    else
                        HexBox.Text = "#" .. string.upper(currentColor:ToHex())
                    end
                end)

                    Size = UDim2.fromOffset(80, 26),
                    Position = UDim2.new(1, -80, 0.5, -13),
                    BackgroundTransparency = 1,
                    Text = "",
                    ZIndex = 5,

                        Popup.Position = UDim2.fromOffset(m.X + 12, m.Y + 12)

                end)

                    Set = function(_, c)
                        hue, sat, val = Color3.toHSV(c)
                        currentColor = c
                        updateAll(true)
                    end,
                    Get = function() return currentColor end,
                    Instance = Container,
                }

    table.insert(Window._conns, toggleConn)

    function Window:Destroy()
        for _, c in ipairs(Window._conns) do
            if typeof(c) == "RBXScriptConnection" and c.Connected then
                c:Disconnect()
            end
        end
        ScreenGui:Destroy()
    end

    Window.ScreenGui = ScreenGui
    Window.Main = Main
    return Window
end

--==============================================================================
-- CONFIG
--==============================================================================
function Library:SaveConfig(name)
    if not writefile then return false, "no writefile" end
    name = name or "rubz_config"
    local ok, json = pcall(HttpService.JSONEncode, HttpService, Library.Flags)
    if not ok then return false, "json encode failed" end
    pcall(writefile, name .. ".json", json)
    return true
end

function Library:LoadConfig(name)
    if not (isfile and readfile) then return false, "no readfile" end
    name = name or "rubz_config"
    if not isfile(name .. ".json") then return false, "file not found" end
    local ok, data = pcall(readfile, name .. ".json")
    if not ok then return false, "read failed" end
    local ok2, tbl = pcall(HttpService.JSONDecode, HttpService, data)
    if not ok2 or type(tbl) ~= "table" then return false, "decode failed" end
    for k, v in pairs(tbl) do Library.Flags[k] = v end
    return true
end

return Library
