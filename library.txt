--// Helpers
local function Create(class, props)
    local inst = Instance.new(class)
    for k, v in pairs(props or {}) do
        if k ~= "Parent" then inst[k] = v end
    end
    if props and props.Parent then inst.Parent = props.Parent end
    return inst
end

local function Corner(inst, r)
    return Create("UICorner", { CornerRadius = UDim.new(0, r or 6), Parent = inst })
end

local function Stroke(inst, color, thick, trans)
    return Create("UIStroke", {
        Color = color or Theme.Outline,
        Thickness = thick or 1,
        Transparency = trans or 0,
        ApplyStrokeMode = Enum.ApplyStrokeMode.Border,
        Parent = inst
    })
end

local function Tween(inst, t, props, style, dir)
    local anim = TweenService:Create(
        inst,
        TweenInfo.new(t or 0.15, style or Enum.EasingStyle.Quad, dir or Enum.EasingDirection.Out),
        props
    )
    anim:Play()
    return anim
end

local function Round(v, range)
    if range <= 5 then return math.floor(v * 100 + 0.5) / 100 end
    return math.floor(v + 0.5)
end

local function GetGuiParent()
    if gethui then
        local ok, hui = pcall(gethui)
        if ok and hui then return hui end
    end
    local ok, cg = pcall(function() return CoreGui end)
    if ok and cg then
        local canUse = pcall(function() return cg.Name end)
        if canUse then return cg end
    end
    return LocalPlayer:WaitForChild("PlayerGui")
end

local function MakeDraggable(frame, handle)
    handle = handle or frame
    local dragging, dragStart, startPos

    handle.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            dragStart = input.Position
            startPos = frame.Position
        end
    end)
    UserInputService.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch then
            dragging = false
        end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if not dragging then return end
        if input.UserInputType == Enum.UserInputType.MouseMovement
        or input.UserInputType == Enum.UserInputType.Touch then
            local delta = input.Position - dragStart
            frame.Position = UDim2.new(
                startPos.X.Scale, startPos.X.Offset + delta.X,
                startPos.Y.Scale, startPos.Y.Offset + delta.Y
            )
        end
    end)
end

local function GetKeyName(input)
    if input.UserInputType == Enum.UserInputType.Keyboard then
        return input.KeyCode.Name
    elseif input.UserInputType == Enum.UserInputType.MouseButton1 then return "Mouse1"
    elseif input.UserInputType == Enum.UserInputType.MouseButton2 then return "Mouse2"
    elseif input.UserInputType == Enum.UserInputType.MouseButton3 then return "Mouse3"
    end
end

function Library:Intro(config)
    config = config or {}
    local duration = config.Duration or 2.6
    local title    = config.Title or "Rubz Library"
    local subtitle = config.Subtitle or ("v" .. Library.Version)
    local parent   = config.Parent or GetGuiParent()

    local ownGui = not config.Gui
    local gui = config.Gui or Create("ScreenGui", {
        Name = "RubzIntro_" .. tostring(math.random(1, 1e6)),
        ResetOnSpawn = false,
        IgnoreGuiInset = true,
        ZIndexBehavior = Enum.ZIndexBehavior.Sibling,
        DisplayOrder = 999,
        Parent = parent,
    })

    local Backdrop = Create("Frame", {
        Size = UDim2.new(1, 0, 1, 0),
        BackgroundColor3 = Color3.fromRGB(0, 0, 0),
        BackgroundTransparency = 1,
        BorderSizePixel = 0,
        Parent = gui,
    })

    local Card = Create("Frame", {
        Size = UDim2.fromOffset(0, 0),
        AnchorPoint = Vector2.new(0.5, 0.5),
        Position = UDim2.new(0.5, 0, 0.5, 0),
        BackgroundColor3 = Theme.Background,
        BorderSizePixel = 0,
        ClipsDescendants = true,
        Parent = gui,
    })
    Corner(Card, 14)
    Stroke(Card, Theme.Accent, 1.5, 0.4)

    local CardGrad = Create("UIGradient", {
        Color = ColorSequence.new({
            ColorSequenceKeypoint.new(0, Color3.fromRGB(20, 30, 60)),
            ColorSequenceKeypoint.new(0.5, Color3.fromRGB(60, 120, 220)),
            ColorSequenceKeypoint.new(1, Color3.fromRGB(20, 30, 60)),
        }),
        Rotation = 0,
        Parent = Card,
    })

    local Title = Create("TextLabel", {
        Size = UDim2.new(1, 0, 0, 30),
        Position = UDim2.new(0, 0, 0.5, -28),
        BackgroundTransparency = 1,
        Text = title,
        TextColor3 = Theme.Text,
        TextSize = 24,
        Font = Enum.Font.GothamBold,
        TextTransparency = 1,
        Parent = Card,
    })
    local Sub = Create("TextLabel", {
        Size = UDim2.new(1, 0, 0, 16),
        Position = UDim2.new(0, 0, 0.5, 2),
        BackgroundTransparency = 1,
        Text = subtitle,
        TextColor3 = Theme.SubText,
        TextSize = 13,
        Font = Enum.Font.Gotham,
        TextTransparency = 1,
        Parent = Card,
    })

    local BarBg = Create("Frame", {
        Size = UDim2.new(0, 220, 0, 4),
        Position = UDim2.new(0.5, -110, 1, -28),
        BackgroundColor3 = Theme.Element,
        BorderSizePixel = 0,
        Parent = Card,
    })
    Corner(BarBg, 2)

    local BarFill = Create("Frame", {
        Size = UDim2.new(0, 0, 1, 0),
        BackgroundColor3 = Theme.Accent,
        BorderSizePixel = 0,
        Parent = BarBg,
    })
    Corner(BarFill, 2)

    Tween(Backdrop, 0.25, { BackgroundTransparency = 0.35 })
    Tween(Card, 0.35, { Size = UDim2.fromOffset(300, 140) }, Enum.EasingStyle.Back)
    task.delay(0.15, function()
        Tween(Title, 0.3, { TextTransparency = 0 })
        Tween(Sub,   0.3, { TextTransparency = 0 })
    end)

    local gradSpin = RunService.RenderStepped:Connect(function(_, dt)
        if CardGrad and CardGrad.Parent then
            CardGrad.Rotation = (CardGrad.Rotation + dt * 60) % 360
        end
    end)

    Tween(BarFill, duration * 0.85, { Size = UDim2.new(1, 0, 1, 0) }, Enum.EasingStyle.Quad)

    task.delay(duration, function()
        if gradSpin then gradSpin:Disconnect() end
        if Card and Card.Parent then
            Tween(Card, 0.3, { Size = UDim2.fromOffset(0, 0) }, Enum.EasingStyle.Back, Enum.EasingDirection.In)
            Tween(Backdrop, 0.35, { BackgroundTransparency = 1 })
            task.delay(0.4, function()
                if ownGui then gui:Destroy() end
            end)
        end
    end)

    return gui
end

        Position = UDim2.new(0.5, -Size.X.Offset / 2, 0.5, -Size.Y.Offset / 2),
        BackgroundColor3 = Theme.Background,
        BorderSizePixel = 0,
        ClipsDescendants = true,
        Parent = ScreenGui,

    Corner(Main, 12)
    Stroke(Main, Theme.Accent, 1, 0.35)

    --// Синий анимированный градиент на фоне
    local BgGradient = Create("UIGradient", {
        Color = ColorSequence.new({
            ColorSequenceKeypoint.new(0.0, Color3.fromRGB(10, 14, 28)),
            ColorSequenceKeypoint.new(0.35, Color3.fromRGB(30, 70, 170)),
            ColorSequenceKeypoint.new(0.5,  Color3.fromRGB(70, 140, 255)),
            ColorSequenceKeypoint.new(0.65, Color3.fromRGB(30, 70, 170)),

        }),
        Rotation = 20,
        Parent = Main,
    })
    -- бегущий "шайн" поверх

        Color = ColorSequence.new({
            ColorSequenceKeypoint.new(0,  Color3.fromRGB(255, 255, 255)),
            ColorSequenceKeypoint.new(0.5, Color3.fromRGB(120, 190, 255)),

            Tween(b, 0.12, { BackgroundColor3 = Theme.Element, TextColor3 = Theme.SubText })

    local NotifHolder = Create("Frame", {

    local currentTab
    local function SelectTab(tab)
        if currentTab == tab then return end
        for _, t in ipairs(Window.Tabs) do
            if t == tab then
                t.Page.Visible = true
                Tween(t.Button, 0.15, { BackgroundTransparency = Library.Opacity.Element, BackgroundColor3 = Theme.Element })
                t.Button.TextColor3 = Theme.Text
                Tween(t.Indicator, 0.15, { BackgroundTransparency = 0 })

                t.Page.Visible = false
                Tween(t.Button, 0.15, { BackgroundTransparency = 1 })

    -- Notify v2 — БЕЗ UIListLayout внутри, с ручной раскладкой
    ---------------------------------------------------------------------------
    local ICONS = { info = "ℹ", success = "✓", warning = "!", error = "✕" }
    local COLOR_MAP = {
        info    = Theme.Info,
        success = Theme.Success,
        warning = Theme.Warning,
        error   = Theme.Danger,
    }

    function Window:Notify(title, text, duration, ntype, icon)
        duration = duration or 4
        ntype = ntype or "info"

        local accentColor, defaultIcon
        if typeof(ntype) == "Color3" then
            accentColor = ntype
            defaultIcon = "•"
        else
            accentColor = COLOR_MAP[ntype] or Theme.Info
            defaultIcon = ICONS[ntype] or "•"
        end
        icon = icon or defaultIcon

        title = tostring(title or "Уведомление")
        text  = tostring(text or "")

        -- размеры
        local WIDTH     = 280
        local PAD       = 12
        local ICON_SIZE = 22
        local TITLE_H   = 18
        local BODY_FONT = Enum.Font.Gotham
        local BODY_SIZE = 12
        local BODY_W    = WIDTH - PAD * 2 - 4

        local bodyH = 0
        if text ~= "" then
            local sz = TextService:GetTextSize(text, BODY_SIZE, BODY_FONT, Vector2.new(BODY_W, 1000))
            bodyH = math.max(sz.Y, 14)
        end
        local HEIGHT = PAD + TITLE_H + (bodyH > 0 and (bodyH + 4) or 0) + PAD

        --// Notif = CanvasGroup (для плавного fade-in)
        local Notif = Create("CanvasGroup", {
            Size = UDim2.fromOffset(WIDTH, HEIGHT),
            BackgroundColor3 = Theme.Panel,
            BackgroundTransparency = 0.05,
            BorderSizePixel = 0,
            GroupTransparency = 1,
            Parent = NotifHolder,
        })
        Corner(Notif, 8)
        Stroke(Notif, accentColor, 1, 0.45)

        -- акцентная полоса слева (абсолютная)
        local Accent = Create("Frame", {
            Size = UDim2.new(0, 3, 1, -16),
            Position = UDim2.new(0, 6, 0, 8),
            BackgroundColor3 = accentColor,
            BorderSizePixel = 0,
            Parent = Notif,
        })
        Corner(Accent, 2)

        -- иконка
        local IconBox = Create("Frame", {
            Size = UDim2.fromOffset(ICON_SIZE, ICON_SIZE),
            Position = UDim2.new(0, 16, 0, PAD - 2),
            BackgroundColor3 = accentColor,
            BackgroundTransparency = 0.82,
            BorderSizePixel = 0,
            Parent = Notif,
        })
        Corner(IconBox, 6)
        Create("TextLabel", {
            Size = UDim2.fromScale(1, 1),
            BackgroundTransparency = 1,
            Text = icon,
            TextColor3 = accentColor,
            TextSize = 13,
            Font = Enum.Font.GothamBold,
            Parent = IconBox,
        })

        -- заголовок
        Create("TextLabel", {
            Size = UDim2.new(1, -(16 + ICON_SIZE + 8), 0, TITLE_H),
            Position = UDim2.new(0, 16 + ICON_SIZE + 8, 0, PAD - 1),
            BackgroundTransparency = 1,
            Text = title,
            TextColor3 = Theme.Text,
            TextSize = 13,
            Font = Enum.Font.GothamBold,
            TextXAlignment = Enum.TextXAlignment.Left,
            TextTruncate = Enum.TextTruncate.AtEnd,
            Parent = Notif,
        })

        -- тело
        if text ~= "" then
            Create("TextLabel", {
                Size = UDim2.new(1, -32, 0, bodyH),
                Position = UDim2.new(0, 16, 0, PAD + TITLE_H + 4),
                BackgroundTransparency = 1,
                Text = text,
                TextColor3 = Theme.SubText,
                TextSize = BODY_SIZE,
                Font = BODY_FONT,
                TextXAlignment = Enum.TextXAlignment.Left,
                TextYAlignment = Enum.TextYAlignment.Top,
                TextWrapped = true,
                Parent = Notif,
            })
        end

        -- fade-in
        Tween(Notif, 0.25, { GroupTransparency = 0 })

        -- auto-hide
        task.delay(duration, function()
            if Notif and Notif.Parent then
                local t = Tween(Notif, 0.25, { GroupTransparency = 1 })
                t.Completed:Connect(function() Notif:Destroy() end)
            end
        end)

        return Notif
    end

            function Section:Slider(text, min, max, default, callback, flag)
                local value = default or min
                local range = max - min
                local Container = Create("Frame", {
                    Size = UDim2.new(1, 0, 0, 44),
                    BackgroundTransparency = 1,
                    Parent = Frame,
                })
                Create("TextLabel", {
                    Size = UDim2.new(1, -80, 0, 18),
                    BackgroundTransparency = 1,
                    Text = text,
                    TextColor3 = Theme.Text,
                    TextSize = 14,
                    Font = Enum.Font.Gotham,
                    TextXAlignment = Enum.TextXAlignment.Left,
                    Parent = Container,
                })
                local ValueLabel = Create("TextLabel", {
                    Size = UDim2.fromOffset(80, 18),
                    Position = UDim2.new(1, -80, 0, 0),
                    BackgroundTransparency = 1,
                    Text = tostring(value),
                    TextColor3 = Theme.SubText,
                    TextSize = 13,
                    Font = Enum.Font.GothamMedium,
                    TextXAlignment = Enum.TextXAlignment.Right,
                    Parent = Container,
                })
                local Bar = Create("Frame", {
                    Size = UDim2.new(1, 0, 0, 6),
                    Position = UDim2.new(0, 0, 0, 32),
                    BackgroundColor3 = Theme.Element,
                    BackgroundTransparency = Library.Opacity.Element,
                    BorderSizePixel = 0,
                    Parent = Container,
                })
                Corner(Bar, 3)
                local Fill = Create("Frame", {
                    Size = UDim2.new(0, 0, 1, 0),
                    BackgroundColor3 = Theme.Accent,
                    BorderSizePixel = 0,
                    Parent = Bar,
                })
                Corner(Fill, 3)
                local Knob = Create("Frame", {
                    Size = UDim2.fromOffset(12, 12),
                    AnchorPoint = Vector2.new(0.5, 0.5),
                    Position = UDim2.new(0, 0, 0.5, 0),
                    BackgroundColor3 = Color3.new(1, 1, 1),
                    BorderSizePixel = 0,
                    ZIndex = 2,
                    Parent = Bar,
                })
                Corner(Knob, 6)
                local Hit = Create("TextButton", {
                    Size = UDim2.new(1, 0, 1, 0),
                    BackgroundTransparency = 1,
                    Text = "",
                    Parent = Container,
                })
                local function apply(alpha, silent)
                    alpha = math.clamp(alpha, 0, 1)
                    value = Round(min + range * alpha, range)
                    Fill.Size = UDim2.new(alpha, 0, 1, 0)
                    Knob.Position = UDim2.new(alpha, 0, 0.5, 0)
                    ValueLabel.Text = tostring(value)
                    if flag then Library.Flags[flag] = value end
                    if not silent and callback then task.spawn(callback, value) end
                end
                local function fromInput(input)
                    local rel = (input.Position.X - Bar.AbsolutePosition.X) / Bar.AbsoluteSize.X
                    apply(rel)
                end
                local dragging = false
                Hit.InputBegan:Connect(function(input)
                    if input.UserInputType == Enum.UserInputType.MouseButton1
                    or input.UserInputType == Enum.UserInputType.Touch then
                        dragging = true
                        fromInput(input)
                    end
                end)
                local c1 = UserInputService.InputChanged:Connect(function(input)
                    if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement
                    or input.UserInputType == Enum.UserInputType.Touch) then
                        fromInput(input)
                    end
                end)
                local c2 = UserInputService.InputEnded:Connect(function(input)
                    if input.UserInputType == Enum.UserInputType.MouseButton1
                    or input.UserInputType == Enum.UserInputType.Touch then
                        dragging = false
                    end
                end)
                table.insert(Window._conns, c1)
                table.insert(Window._conns, c2)
                do
                    local alpha = range > 0 and (value - min) / range or 0
                    Fill.Size = UDim2.new(alpha, 0, 1, 0)
                    Knob.Position = UDim2.new(alpha, 0, 0.5, 0)
                    ValueLabel.Text = tostring(value)
                    if flag then Library.Flags[flag] = value end
                end
                return {
                    Set = function(_, v)
                        local alpha = range > 0 and (v - min) / range or 0
                        apply(alpha, true)
                    end,
                    Get = function() return value end,
                    Instance = Container,
                }
            end

            function Section:Keybind(text, default, callback, flag)
                local currentKey = default or "None"
                local listening = false
                local Container = Create("Frame", {
                    Size = UDim2.new(1, 0, 0, 32),
                    BackgroundTransparency = 1,
                    Parent = Frame,
                })
                Create("TextLabel", {
                    Size = UDim2.new(1, -90, 1, 0),
                    BackgroundTransparency = 1,
                    Text = text,
                    TextColor3 = Theme.Text,
                    TextSize = 14,
                    Font = Enum.Font.Gotham,
                    TextXAlignment = Enum.TextXAlignment.Left,
                    Parent = Container,
                })
                local KeyBtn = Create("TextButton", {
                    Size = UDim2.fromOffset(80, 26),
                    Position = UDim2.new(1, -80, 0.5, -13),
                    BackgroundColor3 = Theme.Element,
                    BackgroundTransparency = Library.Opacity.Element,
                    Text = tostring(currentKey),
                    TextColor3 = Theme.Text,
                    TextSize = 13,
                    Font = Enum.Font.GothamMedium,
                    AutoButtonColor = false,
                    BorderSizePixel = 0,
                    Parent = Container,
                })
                Corner(KeyBtn, 6)
                Stroke(KeyBtn, Theme.Outline, 1, 0.5)
                if flag then Library.Flags[flag] = currentKey end
                KeyBtn.MouseButton1Click:Connect(function()
                    if listening then return end
                    listening = true
                    KeyBtn.Text = "..."
                    Tween(KeyBtn, 0.15, { BackgroundColor3 = Theme.ElementHover })
                end)
                local conn = UserInputService.InputBegan:Connect(function(input, gpe)
                    if listening then
                        if input.UserInputType == Enum.UserInputType.Keyboard
                        or input.UserInputType == Enum.UserInputType.MouseButton1
                        or input.UserInputType == Enum.UserInputType.MouseButton2
                        or input.UserInputType == Enum.UserInputType.MouseButton3 then
                            local name = GetKeyName(input) or "None"
                            if name == "Escape" then name = "None" end
                            currentKey = name
                            KeyBtn.Text = name
                            listening = false
                            Tween(KeyBtn, 0.15, { BackgroundColor3 = Theme.Element })
                            if flag then Library.Flags[flag] = currentKey end
                            if callback then task.spawn(callback, currentKey) end
                        end
                        return
                    end
                    if gpe then return end
                    if currentKey == "None" then return end
                    local name = GetKeyName(input)
                    if name and name == currentKey then
                        if callback then task.spawn(callback, currentKey) end
                    end
                end)
                table.insert(Window._conns, conn)
                return {
                    Set = function(_, v)
                        currentKey = v
                        KeyBtn.Text = tostring(v)
                        if flag then Library.Flags[flag] = v end
                    end,
                    Get = function() return currentKey end,
                    Instance = Container,
                }
            end

            return Section
        end

        table.insert(Window.Tabs, Tab)
        if not currentTab then SelectTab(Tab) end
        return Tab
    end

    --// UI toggle hotkey
    local uiVisible = true
    local toggleKey = config.ToggleKey or Enum.KeyCode.RightShift
    local toggleConn = UserInputService.InputBegan:Connect(function(input, gpe)
        if gpe then return end
        if input.KeyCode == toggleKey then
            uiVisible = not uiVisible
            ScreenGui.Enabled = uiVisible
        end
    end)
    table.insert(Window._conns, toggleConn)

    function Window:Destroy()
        for _, c in ipairs(Window._conns) do
            if typeof(c) == "RBXScriptConnection" and c.Connected then
                c:Disconnect()
            end
        end
        ScreenGui:Destroy()
    end

    Window.ScreenGui = ScreenGui
    Window.Main = Main

    return Window
end
