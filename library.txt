--[[
    SimpleUI v1.0.0
    Лёгкая GUI-библиотека для Roblox-скриптов.

    Возможности:
      • Окно с драгом, сворачиванием и закрытием
      • Табы, секции, лейблы, разделители

      • Уведомления
      • Хранение значений в Library.Flags

    Пример:

        local Window = Library:CreateWindow({ Title = "My Script", Subtitle = "v1.0" })
        local Tab = Window:Tab("Main")
        local Sec = Tab:Section("General")
        Sec:Button("Click me", function() print("hi") end)
        Sec:Toggle("Enable", false, function(v) print(v) end, "enable_flag")
]]

--// Helpers
local function Create(class, props)
    local inst = Instance.new(class)
    for k, v in pairs(props or {}) do
        if k ~= "Parent" then
            inst[k] = v
        end
    end
    if props and props.Parent then
        inst.Parent = props.Parent
    end
    return inst
end

local function Corner(inst, r)
    return Create("UICorner", { CornerRadius = UDim.new(0, r or 6), Parent = inst })
end

local function Stroke(inst, color, thick, trans)
    return Create("UIStroke", {
        Color = color or Theme.Outline,
        Thickness = thick or 1,
        Transparency = trans or 0,
        ApplyStrokeMode = Enum.ApplyStrokeMode.Border,
        Parent = inst
    })
end

local function Tween(inst, t, props, style, dir)
    local anim = TweenService:Create(
        inst,
        TweenInfo.new(t or 0.15, style or Enum.EasingStyle.Quad, dir or Enum.EasingDirection.Out),
        props
    )
    anim:Play()
    return anim
end

local function Round(v, range)
    if range <= 5 then
        return math.floor(v * 100 + 0.5) / 100
    end
    return math.floor(v + 0.5)
end

local function GetGuiParent()
    if gethui then
        local ok, hui = pcall(gethui)
        if ok and hui then return hui end
    end
    local ok, cg = pcall(function() return CoreGui end)
    if ok and cg then
        local canUse = pcall(function() return cg.Name end)
        if canUse then return cg end
    end
    return LocalPlayer:WaitForChild("PlayerGui")
end

local function MakeDraggable(frame, handle)
    handle = handle or frame
    local dragging, dragStart, startPos

    handle.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            dragStart = input.Position
            startPos = frame.Position
        end
    end)

    UserInputService.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch then
            dragging = false
        end
    end)

    UserInputService.InputChanged:Connect(function(input)
        if not dragging then return end
        if input.UserInputType == Enum.UserInputType.MouseMovement
        or input.UserInputType == Enum.UserInputType.Touch then
            local delta = input.Position - dragStart
            frame.Position = UDim2.new(
                startPos.X.Scale, startPos.X.Offset + delta.X,
                startPos.Y.Scale, startPos.Y.Offset + delta.Y
            )
        end
    end)
end

local function GetKeyName(input)
    if input.UserInputType == Enum.UserInputType.Keyboard then
        return input.KeyCode.Name
    elseif input.UserInputType == Enum.UserInputType.MouseButton1 then
        return "Mouse1"
    elseif input.UserInputType == Enum.UserInputType.MouseButton2 then
        return "Mouse2"
    elseif input.UserInputType == Enum.UserInputType.MouseButton3 then
        return "Mouse3"
    end
    return nil
end

        Position = UDim2.new(0.5, -Size.X.Offset / 2, 0.5, -Size.Y.Offset / 2),
        BackgroundColor3 = Theme.Background,
        BorderSizePixel = 0,
        Parent = ScreenGui

    Stroke(Main, Theme.Outline, 1, 0.35)

        Name = "Topbar",
        Size = UDim2.new(1, 0, 0, 42),
        BackgroundColor3 = Theme.Panel,

        Parent = Main

    Create("Frame", {
        Size = UDim2.new(1, 0, 0, 12),
        Position = UDim2.new(0, 0, 1, -12),
        BackgroundColor3 = Theme.Panel,
        BorderSizePixel = 0,
        Parent = Topbar
    })

    Create("TextLabel", {
        Size = UDim2.new(0, 300, 0, 20),
        Position = UDim2.new(0, 16, 0, 6),

        Text = Title,
        TextColor3 = Theme.Text,
        TextSize = 15,
        Font = Enum.Font.GothamBold,

        Parent = Topbar

        Size = UDim2.new(0, 300, 0, 14),
        Position = UDim2.new(0, 16, 0, 24),

            Font = Enum.Font.GothamBold,
            AutoButtonColor = false,
            BorderSizePixel = 0,

        })

        b.MouseEnter:Connect(function()
            Tween(b, 0.12, { BackgroundColor3 = color or Theme.ElementHover, TextColor3 = Theme.Text })
        end)
        b.MouseLeave:Connect(function()
            Tween(b, 0.12, { BackgroundColor3 = Theme.Element, TextColor3 = Theme.SubText })
        end)
        return b
    end

    local CloseBtn    = TopButton("✕", -36, Theme.Danger)

    --// Logic
    local currentTab

    local function SelectTab(tab)
        if currentTab == tab then return end
        for _, t in ipairs(Window.Tabs) do
            if t == tab then
                t.Page.Visible = true
                Tween(t.Button, 0.15, { BackgroundTransparency = 0, BackgroundColor3 = Theme.Element })
                t.Button.TextColor3 = Theme.Text
                Tween(t.Indicator, 0.15, { BackgroundTransparency = 0 })
            else
                t.Page.Visible = false
                Tween(t.Button, 0.15, { BackgroundTransparency = 1 })
                t.Button.TextColor3 = Theme.SubText
                Tween(t.Indicator, 0.15, { BackgroundTransparency = 1 })
            end
        end
        currentTab = tab
    end

    MinimizeBtn.MouseButton1Click:Connect(function()
        minimized = not minimized
        if minimized then
            savedSize = Main.Size
            Tween(Main, 0.2, { Size = UDim2.fromOffset(savedSize.X.Offset, 42) })
        else
            Tween(Main, 0.2, { Size = savedSize })
        end
    end)

    CloseBtn.MouseButton1Click:Connect(function()
        Window:Destroy()
    end)

            function Section:Toggle(text, default, callback, flag)
                local state = default and true or false

                local Container = Create("Frame", {
                    Size = UDim2.new(1, 0, 0, 32),
                    BackgroundTransparency = 1,
                    Parent = Frame
                })

                Create("TextLabel", {
                    Size = UDim2.new(1, -50, 1, 0),
                    BackgroundTransparency = 1,
                    Text = text,
                    TextColor3 = Theme.Text,
                    TextSize = 14,
                    Font = Enum.Font.Gotham,
                    TextXAlignment = Enum.TextXAlignment.Left,
                    Parent = Container
                })

                local Switch = Create("Frame", {
                    Size = UDim2.fromOffset(38, 20),
                    Position = UDim2.new(1, -38, 0.5, -10),
                    BackgroundColor3 = state and Theme.Accent or Theme.Element,
                    BorderSizePixel = 0,
                    Parent = Container
                })
                Corner(Switch, 10)

                local Knob = Create("Frame", {
                    Size = UDim2.fromOffset(14, 14),
                    Position = state and UDim2.new(1, -17, 0.5, -7) or UDim2.new(0, 3, 0.5, -7),
                    BackgroundColor3 = Color3.new(1, 1, 1),
                    BorderSizePixel = 0,
                    Parent = Switch
                })
                Corner(Knob, 7)

                local Hit = Create("TextButton", {
                    Size = UDim2.new(1, 0, 1, 0),
                    BackgroundTransparency = 1,
                    Text = "",
                    Parent = Container
                })

                local function set(value, silent)
                    state = value and true or false
                    Tween(Switch, 0.15, { BackgroundColor3 = state and Theme.Accent or Theme.Element })
                    Tween(Knob, 0.15, {
                        Position = state and UDim2.new(1, -17, 0.5, -7) or UDim2.new(0, 3, 0.5, -7)
                    })
                    if flag then Library.Flags[flag] = state end
                    if not silent and callback then task.spawn(callback, state) end
                end

                Hit.MouseButton1Click:Connect(function() set(not state) end)

                if flag then Library.Flags[flag] = state end

                return {
                    Set = function(_, v) set(v, true) end,
                    Get = function() return state end,
                    Instance = Container,
                }
            end

            --// Slider
            function Section:Slider(text, min, max, default, callback, flag)
                local value = default or min
                local range = max - min

                local Container = Create("Frame", {
                    Size = UDim2.new(1, 0, 0, 44),
                    BackgroundTransparency = 1,
                    Parent = Frame
                })

                Create("TextLabel", {
                    Size = UDim2.new(1, -80, 0, 18),
                    BackgroundTransparency = 1,
                    Text = text,
                    TextColor3 = Theme.Text,
                    TextSize = 14,
                    Font = Enum.Font.Gotham,
                    TextXAlignment = Enum.TextXAlignment.Left,
                    Parent = Container
                })

                local ValueLabel = Create("TextLabel", {
                    Size = UDim2.fromOffset(80, 18),
                    Position = UDim2.new(1, -80, 0, 0),
                    BackgroundTransparency = 1,
                    Text = tostring(value),
                    TextColor3 = Theme.SubText,
                    TextSize = 13,
                    Font = Enum.Font.GothamMedium,
                    TextXAlignment = Enum.TextXAlignment.Right,
                    Parent = Container
                })

                local Bar = Create("Frame", {
                    Size = UDim2.new(1, 0, 0, 6),
                    Position = UDim2.new(0, 0, 0, 32),
                    BackgroundColor3 = Theme.Element,
                    BorderSizePixel = 0,
                    Parent = Container
                })
                Corner(Bar, 3)

                local Fill = Create("Frame", {
                    Size = UDim2.new(0, 0, 1, 0),
                    BackgroundColor3 = Theme.Accent,
                    BorderSizePixel = 0,
                    Parent = Bar
                })
                Corner(Fill, 3)

                local Knob = Create("Frame", {
                    Size = UDim2.fromOffset(12, 12),
                    AnchorPoint = Vector2.new(0.5, 0.5),
                    Position = UDim2.new(0, 0, 0.5, 0),
                    BackgroundColor3 = Color3.new(1, 1, 1),
                    BorderSizePixel = 0,
                    ZIndex = 2,
                    Parent = Bar
                })
                Corner(Knob, 6)

                local Hit = Create("TextButton", {
                    Size = UDim2.new(1, 0, 1, 0),
                    BackgroundTransparency = 1,
                    Text = "",
                    Parent = Container
                })

                local function apply(alpha, silent)
                    alpha = math.clamp(alpha, 0, 1)
                    value = Round(min + range * alpha, range)
                    Fill.Size = UDim2.new(alpha, 0, 1, 0)
                    Knob.Position = UDim2.new(alpha, 0, 0.5, 0)
                    ValueLabel.Text = tostring(value)
                    if flag then Library.Flags[flag] = value end
                    if not silent and callback then task.spawn(callback, value) end
                end

                local function fromInput(input)
                    local rel = (input.Position.X - Bar.AbsolutePosition.X) / Bar.AbsoluteSize.X
                    apply(rel)
                end

                local dragging = false
                Hit.InputBegan:Connect(function(input)
                    if input.UserInputType == Enum.UserInputType.MouseButton1
                    or input.UserInputType == Enum.UserInputType.Touch then
                        dragging = true
                        fromInput(input)
                    end
                end)
                local conn
                conn = UserInputService.InputChanged:Connect(function(input)
                    if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement
                    or input.UserInputType == Enum.UserInputType.Touch) then
                        fromInput(input)
                    end
                end)
                local conn2
                conn2 = UserInputService.InputEnded:Connect(function(input)
                    if input.UserInputType == Enum.UserInputType.MouseButton1
                    or input.UserInputType == Enum.UserInputType.Touch then
                        dragging = false
                    end
                end)
                table.insert(Window._conns, conn)
                table.insert(Window._conns, conn2)

                -- init
                do
                    local alpha = range > 0 and (value - min) / range or 0
                    Fill.Size = UDim2.new(alpha, 0, 1, 0)
                    Knob.Position = UDim2.new(alpha, 0, 0.5, 0)
                    ValueLabel.Text = tostring(value)
                    if flag then Library.Flags[flag] = value end
                end

                return {
                    Set = function(_, v)
                        local alpha = range > 0 and (v - min) / range or 0
                        apply(alpha, true)
                    end,
                    Get = function() return value end,
                    Instance = Container,
                }
            end

                    Text = "▼",
                    TextColor3 = Theme.SubText,
                    TextSize = 10,
                    Font = Enum.Font.GothamBold,
                    Parent = Header
                })

                local OptionsHolder = Create("Frame", {
                    Size = UDim2.new(1, 0, 0, 0),
                    Position = UDim2.new(0, 0, 0, 36),
                    BackgroundTransparency = 1,
                    Parent = Container
                })
                Create("UIListLayout", {
                    Padding = UDim.new(0, 2),
                    SortOrder = Enum.SortOrder.LayoutOrder,
                    Parent = OptionsHolder
                })

                local optionButtons = {}

                local function setValue(v, silent)
                    selected = v
                    SelectedLabel.Text = tostring(v)
                    if flag then Library.Flags[flag] = v end
                    if not silent and callback then task.spawn(callback, v) end
                end

                local function close()
                    if not open then return end
                    open = false
                    Tween(Container, 0.2, { Size = UDim2.new(1, 0, 0, 34) })
                    Tween(Arrow, 0.2, { Rotation = 0 })
                end

                local function openList()
                    if open then return end
                    open = true
                    local h = #options * 28 + 6
                    Tween(Container, 0.2, { Size = UDim2.new(1, 0, 0, 34 + h) })
                    Tween(Arrow, 0.2, { Rotation = 180 })
                end

                for i, opt in ipairs(options) do
                    local OptBtn = Create("TextButton", {
                        Size = UDim2.new(1, -8, 0, 26),
                        Position = UDim2.new(0, 4, 0, 0),
                        BackgroundColor3 = Theme.ElementHover,
                        BackgroundTransparency = 1,
                        Text = "   " .. tostring(opt),
                        TextColor3 = Theme.SubText,
                        TextSize = 13,
                        Font = Enum.Font.Gotham,
                        TextXAlignment = Enum.TextXAlignment.Left,
                        AutoButtonColor = false,
                        BorderSizePixel = 0,
                        Parent = OptionsHolder
                    })
                    Corner(OptBtn, 4)

                    OptBtn.MouseEnter:Connect(function()
                        Tween(OptBtn, 0.1, { BackgroundTransparency = 0, TextColor3 = Theme.Text })
                    end)
                    OptBtn.MouseLeave:Connect(function()
                        Tween(OptBtn, 0.1, { BackgroundTransparency = 1, TextColor3 = Theme.SubText })
                    end)
                    OptBtn.MouseButton1Click:Connect(function()
                        setValue(opt)
                        close()
                    end)

                    optionButtons[i] = OptBtn
                end

                Header.MouseButton1Click:Connect(function()
                    if open then close() else openList() end
                end)

                if flag then Library.Flags[flag] = selected end

                return {
                    Set = function(_, v) setValue(v, true) end,
                    Get = function() return selected end,
                    Refresh = function(_, newOptions)
                        options = newOptions or options
                        for _, b in ipairs(optionButtons) do b:Destroy() end
                        optionButtons = {}
                        for i, opt in ipairs(options) do
                            local OptBtn = Create("TextButton", {
                                Size = UDim2.new(1, -8, 0, 26),
                                Position = UDim2.new(0, 4, 0, 0),
                                BackgroundColor3 = Theme.ElementHover,
                                BackgroundTransparency = 1,
                                Text = "   " .. tostring(opt),
                                TextColor3 = Theme.SubText,
                                TextSize = 13,
                                Font = Enum.Font.Gotham,
                                TextXAlignment = Enum.TextXAlignment.Left,
                                AutoButtonColor = false,
                                BorderSizePixel = 0,
                                Parent = OptionsHolder
                            })
                            Corner(OptBtn, 4)
                            OptBtn.MouseEnter:Connect(function()
                                Tween(OptBtn, 0.1, { BackgroundTransparency = 0, TextColor3 = Theme.Text })
                            end)
                            OptBtn.MouseLeave:Connect(function()
                                Tween(OptBtn, 0.1, { BackgroundTransparency = 1, TextColor3 = Theme.SubText })
                            end)
                            OptBtn.MouseButton1Click:Connect(function()
                                setValue(opt)
                                close()
                            end)
                            optionButtons[i] = OptBtn
                        end
                        if open then
                            Tween(Container, 0.15, { Size = UDim2.new(1, 0, 0, 34 + #options * 28 + 6) })
                        end
                    end,
                    Instance = Container,
                }
            end

            --// Keybind
            function Section:Keybind(text, default, callback, flag)
                local currentKey = default or "None"
                local listening = false

                local Container = Create("Frame", {
                    Size = UDim2.new(1, 0, 0, 32),
                    BackgroundTransparency = 1,
                    Parent = Frame
                })

                Create("TextLabel", {
                    Size = UDim2.new(1, -90, 1, 0),
                    BackgroundTransparency = 1,
                    Text = text,
                    TextColor3 = Theme.Text,
                    TextSize = 14,
                    Font = Enum.Font.Gotham,
                    TextXAlignment = Enum.TextXAlignment.Left,
                    Parent = Container
                })

                local KeyBtn = Create("TextButton", {
                    Size = UDim2.fromOffset(80, 26),
                    Position = UDim2.new(1, -80, 0.5, -13),
                    BackgroundColor3 = Theme.Element,
                    Text = tostring(currentKey),
                    TextColor3 = Theme.Text,
                    TextSize = 13,
                    Font = Enum.Font.GothamMedium,
                    AutoButtonColor = false,
                    BorderSizePixel = 0,
                    Parent = Container
                })
                Corner(KeyBtn, 6)
                Stroke(KeyBtn, Theme.Outline, 1, 0.5)

                if flag then Library.Flags[flag] = currentKey end

                KeyBtn.MouseButton1Click:Connect(function()
                    if listening then return end
                    listening = true
                    KeyBtn.Text = "..."
                    Tween(KeyBtn, 0.15, { BackgroundColor3 = Theme.ElementHover })
                end)

                local conn
                conn = UserInputService.InputBegan:Connect(function(input, gpe)
                    if listening then
                        if input.UserInputType == Enum.UserInputType.Keyboard
                        or input.UserInputType == Enum.UserInputType.MouseButton1
                        or input.UserInputType == Enum.UserInputType.MouseButton2
                        or input.UserInputType == Enum.UserInputType.MouseButton3 then
                            local name = GetKeyName(input) or "None"
                            if name == "Escape" then name = "None" end
                            currentKey = name
                            KeyBtn.Text = name
                            listening = false
                            Tween(KeyBtn, 0.15, { BackgroundColor3 = Theme.Element })
                            if flag then Library.Flags[flag] = currentKey end
                            if callback then task.spawn(callback, currentKey) end
                        end
                        return
                    end

                    if gpe then return end
                    if currentKey == "None" then return end
                    local name = GetKeyName(input)
                    if name and name == currentKey then
                        if callback then task.spawn(callback, currentKey) end
                    end
                end)
                table.insert(Window._conns, conn)

                return {
                    Set = function(_, v)
                        currentKey = v
                        KeyBtn.Text = tostring(v)
                        if flag then Library.Flags[flag] = v end
                    end,
                    Get = function() return currentKey end,
                    Instance = Container,
                }
            end

            return Section
        end

        table.insert(Window.Tabs, Tab)
        if not currentTab then
            SelectTab(Tab)
        end

        return Tab
    end

    ---------------------------------------------------------------------------
    -- Notification
    ---------------------------------------------------------------------------
    function Window:Notify(title, text, duration)
        duration = duration or 4

        local Notif = Create("Frame", {
            Size = UDim2.new(1, 0, 0, 0),
            AutomaticSize = Enum.AutomaticSize.Y,
            BackgroundColor3 = Theme.Panel,
            BorderSizePixel = 0,
            Parent = NotifHolder
        })
        Corner(Notif, 8)
        Stroke(Notif, Theme.Outline, 1, 0.4)

        Create("UIPadding", {
            PaddingTop = UDim.new(0, 10),
            PaddingBottom = UDim.new(0, 10),
            PaddingLeft = UDim.new(0, 14),
            PaddingRight = UDim.new(0, 12),
            Parent = Notif
        })
        Create("UIListLayout", {
            Padding = UDim.new(0, 3),
            SortOrder = Enum.SortOrder.LayoutOrder,
            Parent = Notif
        })

        local Accent = Create("Frame", {
            Size = UDim2.new(0, 3, 1, -12),
            Position = UDim2.new(0, 0, 0, 6),
            BackgroundColor3 = Theme.Accent,
            BorderSizePixel = 0,
            Parent = Notif
        })
        Corner(Accent, 2)

        Create("TextLabel", {
            Size = UDim2.new(1, -10, 0, 16),
            BackgroundTransparency = 1,
            Text = tostring(title or "Уведомление"),
            TextColor3 = Theme.Text,
            TextSize = 13,
            Font = Enum.Font.GothamBold,
            TextXAlignment = Enum.TextXAlignment.Left,
            Parent = Notif
        })

        Create("TextLabel", {
            Size = UDim2.new(1, -10, 0, 0),
            AutomaticSize = Enum.AutomaticSize.Y,
            BackgroundTransparency = 1,
            Text = tostring(text or ""),
            TextColor3 = Theme.SubText,
            TextSize = 12,
            Font = Enum.Font.Gotham,
            TextXAlignment = Enum.TextXAlignment.Left,
            TextWrapped = true,
            Parent = Notif
        })

        -- анимация появления
        Notif.Position = UDim2.new(1, 40, 0, 0)
        Tween(Notif, 0.25, { Position = UDim2.new(0, 0, 0, 0) }, Enum.EasingStyle.Quart)

        task.delay(duration, function()
            if Notif and Notif.Parent then
                local t = Tween(Notif, 0.25, { Position = UDim2.new(1, 60, 0, 0) }, Enum.EasingStyle.Quart)
                t.Completed:Connect(function()
                    Notif:Destroy()
                end)
            end
        end)

        return Notif
    end

    ---------------------------------------------------------------------------
    -- Toggle UI (скрыть/показать)
    ---------------------------------------------------------------------------
    local uiVisible = true
    local toggleKey = config.ToggleKey or Enum.KeyCode.RightShift
    local toggleConn
    toggleConn = UserInputService.InputBegan:Connect(function(input, gpe)
        if gpe then return end
        if input.KeyCode == toggleKey then
            uiVisible = not uiVisible
            ScreenGui.Enabled = uiVisible
        end
    end)
    table.insert(Window._conns, toggleConn)

    ---------------------------------------------------------------------------
    -- Destroy
    ---------------------------------------------------------------------------
    function Window:Destroy()
        for _, c in ipairs(Window._conns) do
            if typeof(c) == "RBXScriptConnection" and c.Connected then
                c:Disconnect()
            end
        end
        ScreenGui:Destroy()
    end

    Window.ScreenGui = ScreenGui
    Window.Main = Main

    return Window
end

return Library
