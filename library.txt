-- ============================================================
-- ВСПОМОГАТЕЛЬНЫЕ ФУНКЦИИ
-- ============================================================
function Rubz.AddCorner(parent, radius)
    local c = Instance.new("UICorner")
    c.CornerRadius = radius or Rubz.Theme.Corner
    c.Parent = parent
    return c
end

function Rubz.AddStroke(parent, color, thickness, transparency)
    local s = Instance.new("UIStroke")
    s.Color = color or Rubz.Theme.Stroke
    s.Thickness = thickness or 1
    s.Transparency = transparency or 0
    s.ApplyStrokeMode = Enum.ApplyStrokeMode.Border
    s.Parent = parent
    return s
end

function Rubz.AddGradient(parent, colorTop, colorBottom, rotation)
    local g = Instance.new("UIGradient")
    g.Color = ColorSequence.new(colorTop, colorBottom)
    g.Rotation = rotation or 90
    g.Parent = parent
    return g
end

function Rubz.AddGlow(parent, color, thickness, transparency)
    local glow = Instance.new("Frame")
    glow.Name = "_Glow"
    glow.BackgroundTransparency = 1
    glow.Size = UDim2.new(1, thickness * 2, 1, thickness * 2)
    glow.Position = UDim2.new(0, -thickness, 0, -thickness)
    glow.ZIndex = parent.ZIndex - 1
    glow.Parent = parent
    Rubz.AddCorner(glow, Rubz.Theme.Corner)
    local s = Rubz.AddStroke(glow, color or Rubz.Theme.Stroke, thickness, transparency or 0.75)
    return glow, s
end

function Rubz.AddShadow(parent, size, transparency)
    local sh = Instance.new("Frame")
    sh.Name = "_Shadow"
    sh.BackgroundColor3 = Color3.new(0, 0, 0)
    sh.BackgroundTransparency = transparency or 0.55
    sh.BorderSizePixel = 0
    sh.Size = UDim2.new(1, size or 6, 1, size or 6)
    sh.Position = UDim2.new(0, -(size or 6) / 2, 0, -(size or 6) / 2 + 3)
    sh.ZIndex = parent.ZIndex - 2
    sh.Parent = parent
    Rubz.AddCorner(sh, Rubz.Theme.Corner)
    return sh
end

function Rubz.Tween(obj, info, props)
    local t = TweenService:Create(obj, info, props)
    t:Play()
    return t
end

function Rubz.PlayClickRipple(button)
    local ripple = Instance.new("Frame")
    ripple.BackgroundColor3 = Rubz.Theme.Accent
    ripple.BackgroundTransparency = 0.6
    ripple.BorderSizePixel = 0
    ripple.AnchorPoint = Vector2.new(0.5, 0.5)
    ripple.Position = UDim2.new(0.5, 0, 0.5, 0)
    ripple.Size = UDim2.new(0, 0, 0, 0)
    ripple.ZIndex = button.ZIndex + 5
    ripple.Parent = button
    Rubz.AddCorner(ripple, Rubz.Theme.Corner)
    local t = TweenService:Create(ripple, TweenInfo.new(0.55, Enum.EasingStyle.Quart, Enum.EasingDirection.Out), {
        Size = UDim2.new(1, 40, 1, 40),
        BackgroundTransparency = 1,
    })
    t:Play()
    t.Completed:Connect(function() ripple:Destroy() end)
end

function Rubz.CreateTab(tabName, iconId)
    local TabsFrame = Rubz.Window.TabsFrame
    local b = Instance.new("TextButton")
    b.Name = "Tab_" .. tabName
    b.Size = UDim2.new(0, 38, 1, 0)
    b.BackgroundColor3 = Rubz.Theme.BgMid
    b.Text = ""
    b.AutoButtonColor = false
    b.Parent = TabsFrame
    Rubz.AddCorner(b)
    Rubz.AddGradient(b, Rubz.Theme.BgSoft, Rubz.Theme.BgDark)
    local s = Rubz.AddStroke(b, Rubz.Theme.StrokeDim, 1, 0.4)

    local indicator = Instance.new("Frame")
    indicator.Name = "_TabGlow"
    indicator.Size = UDim2.new(0, 14, 0, 3)
    indicator.Position = UDim2.new(0.5, -7, 1, -4)
    indicator.BackgroundColor3 = Rubz.Theme.Accent
    indicator.BorderSizePixel = 0
    indicator.Visible = false
    indicator.ZIndex = 5
    indicator.Parent = b
    Rubz.AddCorner(indicator, UDim.new(1, 0))

    local icn = Instance.new("ImageLabel")
    icn.Size = UDim2.new(0, 22, 0, 22)
    icn.Position = UDim2.new(0.5, -11, 0.5, -11)
    icn.BackgroundTransparency = 1
    icn.Image = iconId or "rbxassetid://78134819718605"
    icn.ScaleType = Enum.ScaleType.Fit
    icn.ZIndex = 3
    icn.Parent = b

    b.MouseEnter:Connect(function()
        if Rubz.ActiveTab ~= tabName then
            Rubz.Tween(b, Rubz.Theme.Fast, {BackgroundColor3 = Rubz.Theme.BgHover})
            s.Color = Rubz.Theme.Accent
            s.Transparency = 0
        end
    end)
    b.MouseLeave:Connect(function()
        if Rubz.ActiveTab ~= tabName then
            Rubz.Tween(b, Rubz.Theme.Fast, {BackgroundColor3 = Rubz.Theme.BgMid})
            s.Color = Rubz.Theme.StrokeDim
            s.Transparency = 0.4
        end
    end)

    b.MouseButton1Click:Connect(function()
        Rubz.PlayClickRipple(b)
        if Rubz.ActiveTab == tabName and Rubz.IsPanelOpen then
            Rubz.ClosePanel()
        else
            Rubz.OpenPanel(tabName, iconId)
        end
    end)

    Rubz.TabButtons = Rubz.TabButtons or {}
    table.insert(Rubz.TabButtons, b)
    return b
end

-- ============================================================
-- ОТКРЫТИЕ / ЗАКРЫТИЕ ПАНЕЛИ
-- ============================================================
function Rubz.OpenPanel(tabName, iconId)
    Rubz.ClearSlideContent()
    Rubz.Window.SlideTitle.Text = string.upper(tabName)
    if iconId then Rubz.Window.SlideIcon.Image = iconId end
    Rubz.ActiveTab = tabName
    Rubz.IsPanelOpen = true
    Rubz.Window.SlidePanel.Position = UDim2.new(0.5, -190, 1, 20)
    Rubz.Tween(Rubz.Window.SlidePanel, Rubz.Theme.Smooth, {Position = UDim2.new(0.5, -190, 1, -500)})
    for _, btn in ipairs(Rubz.TabButtons) do
        local isThis = btn.Name == "Tab_" .. tabName
        Rubz.Tween(btn, Rubz.Theme.Fast, {BackgroundColor3 = isThis and Rubz.Theme.AccentDeep or Rubz.Theme.BgMid})
        local glow = btn:FindFirstChild("_TabGlow")
        if glow then glow.Visible = isThis end
    end
end

function Rubz.ClosePanel()
    Rubz.IsPanelOpen = false
    Rubz.ActiveTab = nil
    Rubz.Tween(Rubz.Window.SlidePanel, Rubz.Theme.Smooth, {Position = UDim2.new(0.5, -190, 1, 20)})
    for _, btn in ipairs(Rubz.TabButtons) do
        Rubz.Tween(btn, Rubz.Theme.Fast, {BackgroundColor3 = Rubz.Theme.BgMid})
        local glow = btn:FindFirstChild("_TabGlow")
        if glow then glow.Visible = false end
    end
end

function Rubz.ClearSlideContent()
    Rubz.CurrentOpenSettings = nil
    Rubz.CurrentOpenButton = nil
    Rubz.Window.SettingsContainer.Visible = false
    for _, ch in ipairs(Rubz.Window.SettingsContainer:GetChildren()) do
        if ch:IsA("TextButton") or ch:IsA("Frame") or ch:IsA("TextBox") then ch:Destroy() end
    end
    for _, ch in ipairs(Rubz.Window.FunctionButtonsRow1:GetChildren()) do
        if ch:IsA("TextButton") then ch:Destroy() end
    end
    for _, ch in ipairs(Rubz.Window.FunctionButtonsRow2:GetChildren()) do
        if ch:IsA("TextButton") then ch:Destroy() end
    end
end

function Rubz.CreateFunctionButton(funcData)
    local parent = funcData.row == 2 and Rubz.Window.FunctionButtonsRow2 or Rubz.Window.FunctionButtonsRow1
    local b = Instance.new("TextButton")
    b.AutomaticSize = Enum.AutomaticSize.X
    b.Size = UDim2.new(0, 0, 1, 0)
    b.BackgroundColor3 = funcData.state and Rubz.Theme.AccentDeep or Rubz.Theme.BgMid
    b.Text = funcData.name
    b.TextColor3 = Rubz.Theme.Text
    b.Font = Enum.Font.GothamBold
    b.TextSize = 12
    b.AutoButtonColor = false
    b.ZIndex = 52
    b.Parent = parent
    b.ClipsDescendants = true
    b:SetAttribute("FuncState", funcData.state or false)
    Rubz.AddCorner(b)
    local s = Rubz.AddStroke(b, Rubz.Theme.StrokeDim, 1, 0.4)
    local p = Instance.new("UIPadding")
    p.PaddingLeft = UDim.new(0, 14)
    p.PaddingRight = UDim.new(0, 14)
    p.Parent = b

    b.MouseEnter:Connect(function()
        if Rubz.CurrentOpenButton ~= b then
            Rubz.Tween(b, Rubz.Theme.Fast, {BackgroundColor3 = funcData.state and Rubz.Theme.Accent or Rubz.Theme.BgHover})
            s.Color = Rubz.Theme.Accent
            s.Transparency = 0
        end
    end)
    b.MouseLeave:Connect(function()
        if Rubz.CurrentOpenButton ~= b then
            Rubz.Tween(b, Rubz.Theme.Fast, {BackgroundColor3 = funcData.state and Rubz.Theme.AccentDeep or Rubz.Theme.BgMid})
            s.Color = Rubz.Theme.StrokeDim
            s.Transparency = 0.4
        end
    end)
    b.MouseButton1Click:Connect(function()
        Rubz.PlayClickRipple(b)
        if funcData.onToggle then
            local ns = not funcData.state
            funcData.state = ns
            b:SetAttribute("FuncState", ns)
            funcData.onToggle(ns)
        end
        if Rubz.CurrentOpenButton == b then
            Rubz.CloseCurrentSettings()
        else
            Rubz.CloseCurrentSettings()
            if funcData.settings and #funcData.settings > 0 then
                local panel = Rubz.CreateSettingsPanel(Rubz.Window.SettingsContainer, funcData)
                Rubz.Window.SettingsContainer.Visible = true
                Rubz.CurrentOpenSettings = panel
                Rubz.CurrentOpenButton = b
                s.Color = Rubz.Theme.Accent
                s.Transparency = 0
                Rubz.Tween(b, Rubz.Theme.Fast, {BackgroundColor3 = Rubz.Theme.AccentDeep})
                return
            end
        end
        Rubz.Tween(b, Rubz.Theme.Fast, {BackgroundColor3 = funcData.state and Rubz.Theme.AccentDeep or Rubz.Theme.BgMid})
        s.Color = Rubz.Theme.StrokeDim
        s.Transparency = 0.4
    end)
    return b
end

function Rubz.CloseCurrentSettings()
    if Rubz.CurrentOpenSettings then
        local old = Rubz.CurrentOpenSettings
        Rubz.Tween(old, Rubz.Theme.Fast, {BackgroundTransparency = 1})
        task.delay(0.15, function() if old then old:Destroy() end end)
        Rubz.CurrentOpenSettings = nil
    end
    if Rubz.CurrentOpenButton then
        local state = Rubz.CurrentOpenButton:GetAttribute("FuncState")
        Rubz.Tween(Rubz.CurrentOpenButton, Rubz.Theme.Fast, {BackgroundColor3 = state and Rubz.Theme.AccentDeep or Rubz.Theme.BgMid})
        Rubz.CurrentOpenButton = nil
    end
    Rubz.Window.SettingsContainer.Visible = false
end

function Rubz.CreateSettingsPanel(parent, funcData)
    local sf = Instance.new("Frame")
    sf.Size = UDim2.new(1, 0, 0, 0)
    sf.AutomaticSize = Enum.AutomaticSize.Y
    sf.BackgroundTransparency = 1
    sf.ZIndex = 55
    sf.Parent = parent
    local sfl = Instance.new("UIListLayout")
    sfl.Padding = UDim.new(0, 6)
    sfl.Parent = sf
    if funcData.settings then
        for _, s in ipairs(funcData.settings) do
            if s.type == "slider" then
                Rubz.CreateSlider(sf, s.name, s.min, s.max, s.default, s.callback)
            elseif s.type == "dropdown" then
                Rubz.CreateDropdown(sf, s.name, s.options, s.default, s.callback)
            elseif s.type == "button" then
                Rubz.CreateButton(sf, s.name, s.callback)
            elseif s.type == "textbox" then
                Rubz.CreateTextBox(sf, s.placeholder, s.callback)
            end
        end
    end
    return sf
end

function Rubz.CreateSlider(parent, labelText, minVal, maxVal, defaultVal, callback)
    local c = Instance.new("Frame")
    c.Size = UDim2.new(1, 0, 0, 52)
    c.BackgroundTransparency = 1
    c.Parent = parent

    local l = Instance.new("TextLabel")
    l.Size = UDim2.new(1, -70, 0, 16)
    l.BackgroundTransparency = 1
    l.Text = labelText
    l.TextColor3 = Rubz.Theme.Text
    l.TextSize = 13
    l.Font = Enum.Font.Gotham
    l.TextXAlignment = Enum.TextXAlignment.Left
    l.ZIndex = 62
    l.Parent = c

    local valLabel = Instance.new("TextLabel")
    valLabel.Size = UDim2.new(0, 60, 0, 16)
    valLabel.Position = UDim2.new(1, -60, 0, 0)
    valLabel.BackgroundTransparency = 1
    valLabel.Text = tostring(defaultVal)
    valLabel.TextColor3 = Rubz.Theme.Accent
    valLabel.TextSize = 13
    valLabel.Font = Enum.Font.GothamBold
    valLabel.TextXAlignment = Enum.TextXAlignment.Right
    valLabel.ZIndex = 62
    valLabel.Parent = c

    local t = Instance.new("Frame")
    t.Size = UDim2.new(1, 0, 0, 6)
    t.Position = UDim2.new(0, 0, 0, 30)
    t.BackgroundColor3 = Rubz.Theme.BgSoft
    t.BorderSizePixel = 0
    t.ZIndex = 62
    t.Parent = c
    Rubz.AddCorner(t)
    local tStroke = Rubz.AddStroke(t, Rubz.Theme.StrokeDim, 1, 0.5)

    local f = Instance.new("Frame")
    f.Size = UDim2.new(0, 0, 1, 0)
    f.BackgroundColor3 = Rubz.Theme.Accent
    f.BorderSizePixel = 0
    f.ZIndex = 63
    f.Parent = t
    Rubz.AddCorner(f)
    Rubz.AddGradient(f, Rubz.Theme.Accent, Rubz.Theme.AccentDeep, 0)

    local k = Instance.new("Frame")
    k.Size = UDim2.new(0, 14, 0, 14)
    k.Position = UDim2.new(0, 0, 0.5, -7)
    k.AnchorPoint = Vector2.new(0.5, 0)
    k.BackgroundColor3 = Color3.new(1, 1, 1)
    k.BorderSizePixel = 0
    k.ZIndex = 64
    k.Parent = t
    Rubz.AddCorner(k)
    Rubz.AddStroke(k, Rubz.Theme.Accent, 2, 0)

    local drag = false
    local function update(x)
        local r = math.clamp((x - t.AbsolutePosition.X) / t.AbsoluteSize.X, 0, 1)
        local v = minVal + (maxVal - minVal) * r
        v = math.floor(v * 10 + 0.5) / 10
        f.Size = UDim2.new(r, 0, 1, 0)
        k.Position = UDim2.new(r, 0, 0.5, -7)
        valLabel.Text = tostring(v)
        if callback then callback(v) end
    end
    local dr = (defaultVal - minVal) / (maxVal - minVal)
    f.Size = UDim2.new(dr, 0, 1, 0)
    k.Position = UDim2.new(dr, 0, 0.5, -7)

    t.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 then
            drag = true
            Rubz.Tween(tStroke, Rubz.Theme.Fast, {Color = Rubz.Theme.Accent, Transparency = 0})
            Rubz.Tween(k, Rubz.Theme.Fast, {Size = UDim2.new(0, 18, 0, 18)})
            update(input.Position.X)
        end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if drag and input.UserInputType == Enum.UserInputType.MouseMovement then
            update(input.Position.X)
        end
    end)
    UserInputService.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 and drag then
            drag = false
            Rubz.Tween(tStroke, Rubz.Theme.Fast, {Color = Rubz.Theme.StrokeDim, Transparency = 0.5})
            Rubz.Tween(k, Rubz.Theme.Fast, {Size = UDim2.new(0, 14, 0, 14)})
        end
    end)
end

function Rubz.CreateDropdown(parent, labelText, options, defaultVal, callback)
    local c = Instance.new("Frame")
    c.Size = UDim2.new(1, 0, 0, 70)
    c.BackgroundTransparency = 1
    c.Parent = parent

    local l = Instance.new("TextLabel")
    l.Size = UDim2.new(1, 0, 0, 16)
    l.BackgroundTransparency = 1
    l.Text = labelText
    l.TextColor3 = Rubz.Theme.Text
    l.TextSize = 13
    l.Font = Enum.Font.Gotham
    l.TextXAlignment = Enum.TextXAlignment.Left
    l.ZIndex = 62
    l.Parent = c

    local sb = Instance.new("TextButton")
    sb.Size = UDim2.new(1, 0, 0, 30)
    sb.Position = UDim2.new(0, 0, 0, 20)
    sb.BackgroundColor3 = Rubz.Theme.BgMid
    sb.Text = defaultVal
    sb.TextColor3 = Rubz.Theme.Text
    sb.TextSize = 13
    sb.Font = Enum.Font.Gotham
    sb.AutoButtonColor = false
    sb.ZIndex = 62
    sb.Parent = c
    Rubz.AddCorner(sb)
    local sss = Rubz.AddStroke(sb, Rubz.Theme.StrokeDim, 1, 0.4)
    local sbp = Instance.new("UIPadding")
    sbp.PaddingLeft = UDim.new(0, 12)
    sbp.PaddingRight = UDim.new(0, 30)
    sbp.Parent = sb

    local arrow = Instance.new("TextLabel")
    arrow.Size = UDim2.new(0, 20, 1, 0)
    arrow.Position = UDim2.new(1, -24, 0, 0)
    arrow.BackgroundTransparency = 1
    arrow.Text = "▾"
    arrow.TextColor3 = Rubz.Theme.Accent
    arrow.TextSize = 12
    arrow.Font = Enum.Font.GothamBold
    arrow.ZIndex = 63
    arrow.Parent = sb

    sb.MouseEnter:Connect(function()
        Rubz.Tween(sb, Rubz.Theme.Fast, {BackgroundColor3 = Rubz.Theme.BgHover})
        sss.Color = Rubz.Theme.Accent
        sss.Transparency = 0
    end)
    sb.MouseLeave:Connect(function()
        Rubz.Tween(sb, Rubz.Theme.Fast, {BackgroundColor3 = Rubz.Theme.BgMid})
        sss.Color = Rubz.Theme.StrokeDim
        sss.Transparency = 0.4
    end)

    local ex = false
    local of = Instance.new("Frame")
    of.Size = UDim2.new(1, 0, 0, #options * 28 + 8)
    of.Position = UDim2.new(0, 0, 0, 52)
    of.BackgroundColor3 = Rubz.Theme.BgMid
    of.BorderSizePixel = 0
    of.Visible = false
    of.ZIndex = 70
    of.Parent = c
    of.BackgroundTransparency = 1
    Rubz.AddCorner(of, Rubz.Theme.CornerSmall)
    Rubz.AddStroke(of, Rubz.Theme.Accent, 1, 0.2)
    local ofpad = Instance.new("UIPadding")
    ofpad.PaddingTop = UDim.new(0, 4)
    ofpad.PaddingBottom = UDim.new(0, 4)
    ofpad.Parent = of
    local ofl = Instance.new("UIListLayout")
    ofl.Padding = UDim.new(0, 2)
    ofl.Parent = of

    for _, opt in ipairs(options) do
        local ob = Instance.new("TextButton")
        ob.Size = UDim2.new(1, -8, 0, 24)
        ob.Position = UDim2.new(0, 4, 0, 0)
        ob.BackgroundColor3 = Rubz.Theme.BgSoft
        ob.Text = opt
        ob.TextColor3 = Rubz.Theme.Text
        ob.TextSize = 12
        ob.Font = Enum.Font.Gotham
        ob.AutoButtonColor = false
        ob.ZIndex = 71
        ob.Parent = of
        Rubz.AddCorner(ob)
        ob.MouseEnter:Connect(function()
            Rubz.Tween(ob, Rubz.Theme.Fast, {BackgroundColor3 = Rubz.Theme.AccentDeep, TextColor3 = Color3.new(1, 1, 1)})
        end)
        ob.MouseLeave:Connect(function()
            Rubz.Tween(ob, Rubz.Theme.Fast, {BackgroundColor3 = Rubz.Theme.BgSoft, TextColor3 = Rubz.Theme.Text})
        end)
        ob.MouseButton1Click:Connect(function()
            sb.Text = opt
            ex = false
            Rubz.Tween(of, Rubz.Theme.Fast, {BackgroundTransparency = 1, Position = UDim2.new(0, 0, 0, 52)})
            task.delay(0.15, function() of.Visible = false end)
            Rubz.Tween(arrow, Rubz.Theme.Fast, {Rotation = 0})
            if callback then callback(opt) end
        end)
    end

    sb.MouseButton1Click:Connect(function()
        ex = not ex
        if ex then
            of.Visible = true
            of.BackgroundTransparency = 1
            of.Position = UDim2.new(0, 0, 0, 52)
            Rubz.Tween(of, Rubz.Theme.Fast, {BackgroundTransparency = 0, Position = UDim2.new(0, 0, 0, 54)})
            Rubz.Tween(arrow, Rubz.Theme.Fast, {Rotation = 180})
        else
            Rubz.Tween(of, Rubz.Theme.Fast, {BackgroundTransparency = 1, Position = UDim2.new(0, 0, 0, 52)})
            task.delay(0.15, function() of.Visible = false end)
            Rubz.Tween(arrow, Rubz.Theme.Fast, {Rotation = 0})
        end
    end)
end

function Rubz.CreateTextBox(parent, placeholder, callback)
    local c = Instance.new("Frame")
    c.Size = UDim2.new(1, 0, 0, 36)
    c.BackgroundTransparency = 1
    c.Parent = parent
    local b = Instance.new("TextBox")
    b.Size = UDim2.new(1, 0, 1, 0)
    b.BackgroundColor3 = Rubz.Theme.BgMid
    b.BorderSizePixel = 0
    b.Text = ""
    b.PlaceholderText = placeholder
    b.PlaceholderColor3 = Rubz.Theme.TextDim
    b.TextColor3 = Rubz.Theme.Text
    b.TextSize = 13
    b.Font = Enum.Font.Gotham
    b.ClearTextOnFocus = false
    b.ZIndex = 62
    b.Parent = c
    Rubz.AddCorner(b)
    local bs = Rubz.AddStroke(b, Rubz.Theme.StrokeDim, 1, 0.4)
    local bp = Instance.new("UIPadding")
    bp.PaddingLeft = UDim.new(0, 12)
    bp.PaddingRight = UDim.new(0, 12)
    bp.Parent = b

    b.FocusLost:Connect(function()
