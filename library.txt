local function Create(class, props)
    local inst = Instance.new(class)
    for k, v in pairs(props or {}) do
        if k ~= "Parent" then inst[k] = v end
    end
    if props and props.Parent then inst.Parent = props.Parent end
    return inst
end

local function Corner(inst, r)
    return Create("UICorner", { CornerRadius = UDim.new(0, r or 6), Parent = inst })
end

local function Stroke(inst, color, thick, trans)
    return Create("UIStroke", {
        Color = color or Theme.Outline,
        Thickness = thick or 1,
        Transparency = trans or 0,
        ApplyStrokeMode = Enum.ApplyStrokeMode.Border,
        Parent = inst
    })
end

local function Tween(inst, t, props, style, dir)
    local anim = TweenService:Create(
        inst,
        TweenInfo.new(t or 0.15, style or Enum.EasingStyle.Quad, dir or Enum.EasingDirection.Out),
        props
    )
    anim:Play()
    return anim
end

local function Round(v, range)
    if range <= 5 then return math.floor(v * 100 + 0.5) / 100 end
    return math.floor(v + 0.5)
end

local function GetGuiParent()
    if gethui then
        local ok, hui = pcall(gethui)
        if ok and hui then return hui end
    end
    local ok, cg = pcall(function() return CoreGui end)
    if ok and cg then
        local canUse = pcall(function() return cg.Name end)
        if canUse then return cg end
    end
    return LocalPlayer:WaitForChild("PlayerGui")
end

local function MakeDraggable(frame, handle)
    handle = handle or frame
    local dragging, dragStart, startPos

    handle.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            dragStart = input.Position
            startPos = frame.Position
        end
    end)
    UserInputService.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch then
            dragging = false
        end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if not dragging then return end
        if input.UserInputType == Enum.UserInputType.MouseMovement
        or input.UserInputType == Enum.UserInputType.Touch then
            local delta = input.Position - dragStart
            frame.Position = UDim2.new(
                startPos.X.Scale, startPos.X.Offset + delta.X,
                startPos.Y.Scale, startPos.Y.Offset + delta.Y
            )
        end
    end)
end

local function GetKeyName(input)
    if input.UserInputType == Enum.UserInputType.Keyboard then
        return input.KeyCode.Name
    elseif input.UserInputType == Enum.UserInputType.MouseButton1 then return "Mouse1"
    elseif input.UserInputType == Enum.UserInputType.MouseButton2 then return "Mouse2"
    elseif input.UserInputType == Enum.UserInputType.MouseButton3 then return "Mouse3"
    end
end

-- Рисует "X" из двух полос (для крестика — избегаем проблем со шрифтом)
local function DrawX(parent, size, thickness, color, zIndex)
    size = size or 12
    thickness = thickness or 2
    color = color or Theme.Text
    local container = Create("Frame", {
        Size = UDim2.fromOffset(size, size),
        AnchorPoint = Vector2.new(0.5, 0.5),
        Position = UDim2.new(0.5, 0, 0.5, 0),
        BackgroundTransparency = 1,
        ZIndex = zIndex or 2,
        Parent = parent,
    })
    local l1 = Create("Frame", {
        Size = UDim2.fromOffset(size, thickness),
        AnchorPoint = Vector2.new(0.5, 0.5),
        Position = UDim2.new(0.5, 0, 0.5, 0),
        BackgroundColor3 = color,
        BorderSizePixel = 0,
        Rotation = 45,
        ZIndex = zIndex or 2,
        Parent = container,
    })
    Corner(l1, thickness / 2)
    local l2 = Create("Frame", {
        Size = UDim2.fromOffset(size, thickness),
        AnchorPoint = Vector2.new(0.5, 0.5),
        Position = UDim2.new(0.5, 0, 0.5, 0),
        BackgroundColor3 = color,
        BorderSizePixel = 0,
        Rotation = -45,
        ZIndex = zIndex or 2,
        Parent = container,
    })
    Corner(l2, thickness / 2)
    return container, l1, l2
end

-- Рисует "—" (минус) одной полосой

    width = width or 12

        BackgroundColor3 = color,

    return line
end

-- Ripple-эффект на кнопке
local function AddRipple(button)
    button.ClipsDescendants = true
    button.MouseButton1Down:Connect(function()
        local mousePos = UserInputService:GetMouseLocation()
        local relX = mousePos.X - button.AbsolutePosition.X
        local relY = mousePos.Y - button.AbsolutePosition.Y

        local ripple = Create("Frame", {
            Size = UDim2.fromOffset(10, 10),
            Position = UDim2.fromOffset(relX - 5, relY - 5),
            BackgroundColor3 = Color3.fromRGB(255, 255, 255),
            BackgroundTransparency = 0.6,
            BorderSizePixel = 0,
            ZIndex = button.ZIndex + 5,
            Parent = button,
        })
        Corner(ripple, 999)

        local size = math.max(button.AbsoluteSize.X, button.AbsoluteSize.Y) * 2.2
        Tween(ripple, 0.55, {
            Size = UDim2.fromOffset(size, size),
            Position = UDim2.fromOffset(relX - size / 2, relY - size / 2),
            BackgroundTransparency = 1,
        }, Enum.EasingStyle.Quart)

        task.delay(0.6, function()
            if ripple and ripple.Parent then ripple:Destroy() end
        end)
    end)
end

-- Tooltip система (одна на окно)
local function CreateTooltipLayer(parentGui)
    local Tip = Create("Frame", {
        Size = UDim2.fromOffset(0, 0),
        AutomaticSize = Enum.AutomaticSize.XY,
        BackgroundColor3 = Color3.fromRGB(15, 20, 38),
        BorderSizePixel = 0,
        Visible = false,
        ZIndex = 9999,
        Parent = parentGui,
    })
    Corner(Tip, 6)
    Stroke(Tip, Theme.Accent, 1, 0.5)
    Create("UIPadding", {
        PaddingTop = UDim.new(0, 4),
        PaddingBottom = UDim.new(0, 4),
        PaddingLeft = UDim.new(0, 8),
        PaddingRight = UDim.new(0, 8),
        Parent = Tip,
    })
    local Label = Create("TextLabel", {
        Size = UDim2.new(0, 0, 0, 0),
        AutomaticSize = Enum.AutomaticSize.XY,
        BackgroundTransparency = 1,
        Text = "",
        TextColor3 = Theme.Text,
        TextSize = 12,
        Font = Enum.Font.Gotham,
        Parent = Tip,
    })
    return { Frame = Tip, Label = Label }
end

        Text = title,

        Rotation = 0,

        Tween(Title, 0.3, { TextTransparency = 0 })

        if spin then spin:Disconnect() end

            Tween(Card, 0.3, { Size = UDim2.fromOffset(0, 0) }, Enum.EasingStyle.Back, Enum.EasingDirection.In)
            Tween(Backdrop, 0.35, { BackgroundTransparency = 1 })
            task.delay(0.4, function()

            end)
        end
    end)

        Position = UDim2.new(0.5, 0, 0.5, 0),
        AnchorPoint = Vector2.new(0.5, 0.5),

        BorderSizePixel = 0,
        ClipsDescendants = true,

    Stroke(Main, Theme.Accent, 1, 0.3)

    local function SelectTab(tab)
        if currentTab == tab then return end

        -- fade out old
        if currentTab and currentTab.Page then
            currentTab.Page.Visible = false
        end

        for _, t in ipairs(Window.Tabs) do
            if t == tab then

                    state = value and true or false

                        BackgroundColor3 = state and Theme.Accent or Theme.Element,

                    }, Enum.EasingStyle.Quart)
                    Tween(Knob, 0.25, {

                    Get = function() return state end,

                }

            -- SLIDER
            function Section:Slider(text, min, max, default, callback, flag)
                local value = default or min
                local range = max - min
                local Container = Create("Frame", {
                    Size = UDim2.new(1, 0, 0, 46),
                    BackgroundTransparency = 1,
                    Parent = Frame,
                })
                Create("TextLabel", {
                    Size = UDim2.new(1, -80, 0, 18),
                    BackgroundTransparency = 1,
                    Text = text,
                    TextColor3 = Theme.Text,
                    TextSize = 14,
                    Font = Enum.Font.Gotham,
                    TextXAlignment = Enum.TextXAlignment.Left,
                    Parent = Container,
                })
                local ValueLabel = Create("TextLabel", {
                    Size = UDim2.fromOffset(80, 18),
                    Position = UDim2.new(1, -80, 0, 0),
                    BackgroundTransparency = 1,
                    Text = tostring(value),
                    TextColor3 = Theme.Accent,
                    TextSize = 13,
                    Font = Enum.Font.GothamBold,
                    TextXAlignment = Enum.TextXAlignment.Right,
                    Parent = Container,
                })
                local Bar = Create("Frame", {
                    Size = UDim2.new(1, 0, 0, 6),
                    Position = UDim2.new(0, 0, 0, 34),
                    BackgroundColor3 = Theme.Element,
                    BackgroundTransparency = Library.Opacity.Element,
                    BorderSizePixel = 0,
                    Parent = Container,
                })
                Corner(Bar, 3)
                local Fill = Create("Frame", {
                    Size = UDim2.new(0, 0, 1, 0),
                    BackgroundColor3 = Theme.Accent,
                    BorderSizePixel = 0,
                    Parent = Bar,
                })
                Corner(Fill, 3)
                local FillGrad = Create("UIGradient", {
                    Color = ColorSequence.new({
                        ColorSequenceKeypoint.new(0, Theme.AccentDark),
                        ColorSequenceKeypoint.new(1, Theme.AccentBright),
                    }),
                    Parent = Fill,
                })

                local Knob = Create("Frame", {
                    Size = UDim2.fromOffset(12, 12),
                    AnchorPoint = Vector2.new(0.5, 0.5),
                    Position = UDim2.new(0, 0, 0.5, 0),
                    BackgroundColor3 = Color3.new(1, 1, 1),
                    BorderSizePixel = 0,
                    ZIndex = 2,
                    Parent = Bar,
                })
                Corner(Knob, 999)
                local KnobStroke = Stroke(Knob, Theme.Accent, 2, 0)

                local Hit = Create("TextButton", {
                    Size = UDim2.new(1, 0, 1, 0),
                    BackgroundTransparency = 1,
                    Text = "",
                    Parent = Container,
                })

                local function apply(alpha, silent)
                    alpha = math.clamp(alpha, 0, 1)
                    value = Round(min + range * alpha, range)
                    Fill.Size = UDim2.new(alpha, 0, 1, 0)
                    Knob.Position = UDim2.new(alpha, 0, 0.5, 0)
                    ValueLabel.Text = tostring(value)
                    if flag then Library.Flags[flag] = value end
                    if not silent and callback then task.spawn(callback, value) end
                end
                local function fromInput(input)
                    local rel = (input.Position.X - Bar.AbsolutePosition.X) / Bar.AbsoluteSize.X
                    apply(rel)
                end

                local dragging = false
                Hit.InputBegan:Connect(function(input)
                    if input.UserInputType == Enum.UserInputType.MouseButton1
                    or input.UserInputType == Enum.UserInputType.Touch then
                        dragging = true
                        fromInput(input)
                        Tween(Knob, 0.1, { Size = UDim2.fromOffset(16, 16) })
                        Tween(Bar, 0.1, { Size = UDim2.new(1, 0, 0, 8), Position = UDim2.new(0, 0, 0, 33) })
                    end
                end)
                local c1 = UserInputService.InputChanged:Connect(function(input)
                    if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement
                    or input.UserInputType == Enum.UserInputType.Touch) then
                        fromInput(input)
                    end
                end)
                local c2 = UserInputService.InputEnded:Connect(function(input)
                    if input.UserInputType == Enum.UserInputType.MouseButton1
                    or input.UserInputType == Enum.UserInputType.Touch then
                        if dragging then
                            dragging = false
                            Tween(Knob, 0.15, { Size = UDim2.fromOffset(12, 12) }, Enum.EasingStyle.Back)
                            Tween(Bar, 0.15, { Size = UDim2.new(1, 0, 0, 6), Position = UDim2.new(0, 0, 0, 34) })
                        end
                    end
                end)
                table.insert(Window._conns, c1)
                table.insert(Window._conns, c2)

                do
                    local alpha = range > 0 and (value - min) / range or 0
                    Fill.Size = UDim2.new(alpha, 0, 1, 0)
                    Knob.Position = UDim2.new(alpha, 0, 0.5, 0)
                    ValueLabel.Text = tostring(value)
                    if flag then Library.Flags[flag] = value end
                end

                return {
                    Set = function(_, v)
                        local alpha = range > 0 and (v - min) / range or 0
                        apply(alpha, true)
                    end,
                    Get = function() return value end,
                    Instance = Container,
                }
            end

                    clearOptions()
                    for i, opt in ipairs(list) do
                        local OptBtn = Create("TextButton", {
                            Size = UDim2.new(1, -8, 0, 26),
                            Position = UDim2.new(0, 4, 0, 0),
                            BackgroundColor3 = Theme.ElementHover,
                            BackgroundTransparency = 1,
                            Text = "   " .. tostring(opt),
                            TextColor3 = Theme.SubText,
                            TextSize = 13,
                            Font = Enum.Font.Gotham,
                            TextXAlignment = Enum.TextXAlignment.Left,
                            AutoButtonColor = false,
                            BorderSizePixel = 0,
                            Parent = OptionsHolder,
                        })
                        Corner(OptBtn, 4)
                        OptBtn.MouseEnter:Connect(function()
                            Tween(OptBtn, 0.1, { BackgroundTransparency = 0, TextColor3 = Theme.Text })
                        end)
                        OptBtn.MouseLeave:Connect(function()
                            Tween(OptBtn, 0.1, { BackgroundTransparency = 1, TextColor3 = Theme.SubText })
                        end)
                        OptBtn.MouseButton1Click:Connect(function()
                            setValue(opt); close()
                        end)
                        optionButtons[i] = OptBtn
                    end
                end
                local function openList()
                    if open then return end
                    open = true
                    local h = math.min(#options, 6) * 28 + 6
                    Tween(Container, 0.25, { Size = UDim2.new(1, 0, 0, 34 + h) }, Enum.EasingStyle.Quart)
                    Tween(Arrow, 0.25, { Rotation = 180 })
                end

                renderOptions(options)

                Header.MouseButton1Click:Connect(function()
                    if open then close() else openList() end
                end)
                if flag then Library.Flags[flag] = selected end

                return {
                    Set = function(_, v) setValue(v, true) end,
                    Get = function() return selected end,
                    Refresh = function(_, newOpts)
                        allOptions = newOpts or allOptions
                        options = allOptions
                        renderOptions(options)
                    end,
                    Instance = Container,
                }
            end

            -- KEYBIND
            function Section:Keybind(text, default, callback, flag)
                local currentKey = default or "None"
                local listening = false
                local Container = Create("Frame", {
                    Size = UDim2.new(1, 0, 0, 34),
                    BackgroundTransparency = 1,
                    Parent = Frame,
                })
                Create("TextLabel", {
                    Size = UDim2.new(1, -90, 1, 0),
                    BackgroundTransparency = 1,
                    Text = text,
                    TextColor3 = Theme.Text,
                    TextSize = 14,
                    Font = Enum.Font.Gotham,
                    TextXAlignment = Enum.TextXAlignment.Left,
                    Parent = Container,
                })
                local KeyBtn = Create("TextButton", {
                    Size = UDim2.fromOffset(80, 28),
                    Position = UDim2.new(1, -80, 0.5, -14),
                    BackgroundColor3 = Theme.Element,
                    BackgroundTransparency = Library.Opacity.Element,
                    Text = tostring(currentKey),
                    TextColor3 = Theme.Text,
                    TextSize = 13,
                    Font = Enum.Font.GothamMedium,
                    AutoButtonColor = false,
                    BorderSizePixel = 0,
                    Parent = Container,
                })
                Corner(KeyBtn, 6)
                local KBStroke = Stroke(KeyBtn, Theme.Outline, 1, 0.6)

                if flag then Library.Flags[flag] = currentKey end
                KeyBtn.MouseButton1Click:Connect(function()
                    if listening then return end
                    listening = true
                    KeyBtn.Text = "..."
                    Tween(KeyBtn, 0.15, { BackgroundColor3 = Theme.Accent, BackgroundTransparency = 0.3 })
                    Tween(KBStroke, 0.15, { Color = Theme.AccentBright, Transparency = 0 })
                end)

                local conn = UserInputService.InputBegan:Connect(function(input, gpe)
                    if listening then
                        if input.UserInputType == Enum.UserInputType.Keyboard
                        or input.UserInputType == Enum.UserInputType.MouseButton1
                        or input.UserInputType == Enum.UserInputType.MouseButton2
                        or input.UserInputType == Enum.UserInputType.MouseButton3 then
                            local name = GetKeyName(input) or "None"
                            if name == "Escape" then name = "None" end
                            currentKey = name
                            KeyBtn.Text = name
                            listening = false
                            Tween(KeyBtn, 0.15, { BackgroundColor3 = Theme.Element, BackgroundTransparency = Library.Opacity.Element })
                            Tween(KBStroke, 0.15, { Color = Theme.Outline, Transparency = 0.6 })
                            if flag then Library.Flags[flag] = currentKey end
                            if callback then task.spawn(callback, currentKey) end
                        end
                        return
                    end
                    if gpe then return end
                    if currentKey == "None" then return end
                    local name = GetKeyName(input)
                    if name and name == currentKey then
                        if callback then task.spawn(callback, currentKey) end
                    end
                end)
                table.insert(Window._conns, conn)

                return {
                    Set = function(_, v)
                        currentKey = v
                        KeyBtn.Text = tostring(v)
                        if flag then Library.Flags[flag] = v end
                    end,
                    Get = function() return currentKey end,
                    Instance = Container,
                }
            end

                    if not fromHex then

                    Preview.BackgroundColor3 = currentColor
                    SVSquare.BackgroundColor3 = Color3.fromHSV(hue, 1, 1)
                    SVKnob.Position = UDim2.new(sat, 0, 1 - val, 0)
                    HueKnob.Position = UDim2.new(hue, 0, 0.5, 0)
                    if flag then Library.Flags[flag] = currentColor end
                    if callback then task.spawn(callback, currentColor) end

                    local relX = math.clamp((input.Position.X - SVSquare.AbsolutePosition.X) / SVSquare.AbsoluteSize.X, 0, 1)
                    local relY = math.clamp((input.Position.Y - SVSquare.AbsolutePosition.Y) / SVSquare.AbsoluteSize.Y, 0, 1)
                    sat = relX
                    val = 1 - relY
                    updateAll()
                end

                    local rel = math.clamp((input.Position.X - HueBar.AbsolutePosition.X) / HueBar.AbsoluteSize.X, 0, 1)
                    hue = rel
                    updateAll()
                end

                        hueFromInput(input)
                    end

                    or input.UserInputType == Enum.UserInputType.Touch then
                        if svDrag then svFromInput(input) end
                        if hueDrag then hueFromInput(input) end
                    end

                    if input.UserInputType == Enum.UserInputType.MouseButton1
                    or input.UserInputType == Enum.UserInputType.Touch then
                        svDrag = false
                        hueDrag = false
                    end
                end)

                table.insert(Window._conns, upConn)

                    local hex = HexBox.Text:gsub("#", "")
                    local ok, col = pcall(Color3.fromHex, hex)
                    if ok then
                        hue, sat, val = Color3.toHSV(col)
                        currentColor = col
                        updateAll(true)
                    else
                        HexBox.Text = "#" .. string.upper(currentColor:ToHex())
                    end
                end)

                local pickerOpen = false
                Preview.MouseButton1Click = nil

                    Size = UDim2.fromOffset(80, 26),
                    Position = UDim2.new(1, -80, 0.5, -13),

                    Text = "",
                    ZIndex = 5,

                    end
                end)

                return {
                    Set = function(_, c)
                        hue, sat, val = Color3.toHSV(c)
                        currentColor = c
                        updateAll(true)
                    end,
                    Get = function() return currentColor end,
                    Instance = Container,
                }
            end

        table.insert(Window.Tabs, Tab)
        if not currentTab then SelectTab(Tab) end

    end

    --// UI toggle
    local uiVisible = true
    local toggleKey = config.ToggleKey or Enum.KeyCode.RightShift
    local toggleConn = UserInputService.InputBegan:Connect(function(input, gpe)
        if gpe then return end
        if input.KeyCode == toggleKey then
            uiVisible = not uiVisible
            ScreenGui.Enabled = uiVisible
        end
    end)
    table.insert(Window._conns, toggleConn)

    --// DESTROY
    function Window:Destroy()
        for _, c in ipairs(Window._conns) do
            if typeof(c) == "RBXScriptConnection" and c.Connected then
                c:Disconnect()
            end
        end
        ScreenGui:Destroy()
    end

    Window.ScreenGui = ScreenGui
    Window.Main = Main

    return Window
end

--==============================================================================
-- CONFIG SAVE / LOAD
--==============================================================================
function Library:SaveConfig(name)
    if not writefile then return false, "no writefile" end
    name = name or "rubz_config"
    local ok, json = pcall(HttpService.JSONEncode, HttpService, Library.Flags)
    if not ok then return false, "json encode failed" end
    pcall(writefile, name .. ".json", json)
    return true
end

function Library:LoadConfig(name)
    if not (isfile and readfile) then return false, "no readfile" end
    name = name or "rubz_config"
    if not isfile(name .. ".json") then return false, "file not found" end
    local ok, data = pcall(readfile, name .. ".json")
    if not ok then return false, "read failed" end
    local ok2, tbl = pcall(HttpService.JSONDecode, HttpService, data)
    if not ok2 or type(tbl) ~= "table" then return false, "decode failed" end
    for k, v in pairs(tbl) do
        Library.Flags[k] = v
    end
    return true
end

return Library
